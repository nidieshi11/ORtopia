\section*{保留的核心模型}
根据\texttt{}公式以及PDF文件，可以直接求解的核心模型是第20页上的线性规划重构形式，用$(R''_{\mathcal K})$表示。第13页上的原始连续时间鲁棒问题$(R)$，以及第19页上的中间鲁棒对应问题$(R'_{\mathcal K})$，由于它们属于非凸或半无限形式，因此被省略了。论文证明，原始的鲁棒V2G问题与下面所保留的线性规划问题是完全等价的。

\paragraph{决策变量。}
对于每一个周期$k \in \mathcal{K}$，$x_k^b$表示基准充电功率，$x_k^r$是规定价格，$m_k$是一个辅助变量，而$z$则是终端分段仿射惩罚的变量。另外的变量$\lambda_k^+,\lambda_k^-,\theta_k^+,\theta_k^-,\Lambda_{k,l}^+,\Lambda_{k,l}^-,
\Theta_{k,l}^+,\Theta_{k,l}^-$是由鲁棒优化方法引入的对偶变量。

\paragraph{参数和索引集。}
可行的充电集合为$\mathcal{X}_{\mathcal K}$。边界条件为$\bar y_k^+,\bar y_k^-,\bar y,\underline y,\bar y_0,\underline y_0,
\hat{\bar y}_0,\hat{\underline y}_0$，即电池容量和初始状态的限制。参数$\eta^+,\eta^-,\Delta t,\gamma,\hat{\gamma},d_k,r_n,q_n$与论文中的设置相同，而$\Delta\eta = \frac{1}{\eta^-} - \eta^+ > 0$则有所不同。终端惩罚由$\mathcal{N}_+,\mathcal{N}_-,\mathcal{N}_0$表示的断点集合来表示。时间范围指标为$I(k,l)=\min\{k,\ l+\Gamma/\Delta t-1\}$和$\hat I(K,k)=\min\{K,\ k+\hat{\Gamma}/\Delta t-1\}$。

\section*{线性规划的重述}
\begin{equation}
\min \quad c_{\mathcal K}(\boldsymbol{x}^b,\boldsymbol{x}^r) + z
\tag{\ensuremath{R''_{\mathcal K}}-Obj}
\end{equation}

\begin{equation}
\boldsymbol{x}^b,\boldsymbol{x}^r \in \mathcal{X}_{\mathcal K},\quad
z \in \mathbb{R},\quad
\boldsymbol{m},\boldsymbol{\lambda}^+,\boldsymbol{\lambda}^-,
\boldsymbol{\theta}^+,\boldsymbol{\theta}^- \in \mathbb{R}_+^K,
\tag{LP-a}
\end{equation}

\begin{equation}
\boldsymbol{\Lambda}^+,\boldsymbol{\Lambda}^-,
\boldsymbol{\Theta}^+,\boldsymbol{\Theta}^- \in \mathbb{R}_+^{K \times K}
\tag{LP-b}
\end{equation}

\begin{equation}
x_k^r + x_k^b \le \bar y_k^+,\qquad
x_k^r - x_k^b \le \bar y_k^-
\quad \forall k \in \mathcal K
\tag{27}
\end{equation}

\begin{equation}
m_k \ge \eta^+ x_k^r,\qquad
m_k \ge \frac{1}{\eta^-}x_k^r - \Delta\eta\,x_k^b
\quad \forall k \in \mathcal K
\tag{28}
\end{equation}

\begin{equation}
\sum_{l=1}^{k} \Delta t \left( \eta^+x_l^b + \Lambda_{k,l}^+ - d_l \right)
\;+\; \gamma \sum_{l=1}^{k} \Theta_{k,l}^+
\le \bar y - \bar y_0
\quad \forall k \in \mathcal K \cup \{0\}
\tag{29}
\end{equation}

\begin{equation}
\sum_{l=1}^{k} \Delta t \left( \eta^+x_l^b - \Lambda_{k,l}^- - d_l \right)
\;-\; \gamma \sum_{l=1}^{k} \Theta_{k,l}^-
\ge \underline y - \underline y_0
\quad \forall k \in \mathcal K \cup \{0\}
\tag{30}
\end{equation}

\begin{equation}
\sum_{k \in \mathcal K} \Delta t \left( \eta^+x_k^b + \lambda_k^+ - d_k \right)
\;+\; \hat{\gamma}\sum_{k \in \mathcal K}\theta_k^+
\le \frac{z - r_n}{q_n} - \hat{\bar y}_0
\quad \forall n \in \mathcal N_+
\tag{31}
\end{equation}

\begin{equation}
\sum_{k \in \mathcal K} \Delta t \left( \eta^+x_k^b - \lambda_k^- - d_k \right)
\;-\; \hat{\gamma}\sum_{k \in \mathcal K}\theta_k^-
\ge \frac{z - r_n}{q_n} - \hat{\underline y}_0
\quad \forall n \in \mathcal N_-
\tag{32}
\end{equation}

\begin{equation}
r_n \le z
\quad \forall n \in \mathcal N_0
\tag{33}
\end{equation}

\begin{equation}
\Lambda_{k,l}^+ + \sum_{i=l}^{I(k,l)} \Theta_{k,i}^+ \ge \eta^+ x_l^r,\qquad
\Lambda_{k,l}^- + \sum_{i=l}^{I(k,l)} \Theta_{k,i}^- \ge m_l
\quad \forall k,l \in \mathcal K:\ l \le k
\tag{34}
\end{equation}

\begin{equation}
\lambda_k^+ + \sum_{i=k}^{\hat I(K,k)} \theta_i^+ \ge \eta^+ x_k^r,\qquad
\lambda_k^- + \sum_{i=k}^{\hat I(K,k)} \theta_i^- \ge m_k
\quad \forall k \in \mathcal K
\tag{35}
\end{equation}
