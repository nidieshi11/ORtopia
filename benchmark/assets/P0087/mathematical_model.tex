\section*{SP-CVaR模型的重新表述}
直接可求解的核心模型，实际上是随机 CVaR 公式的采样平均近似模型。它保留了原始的第一阶段分配和排序决策，同时包含了基于情景的补偿变量。

\begin{equation}
\begin{aligned}
\min \quad
& \sum_{r \in R} f_r v_r
+ \sum_{a \in A} f_a y_a
+ \tau
+ \frac{1}{N(1-\gamma)} \sum_{n=1}^N \eta_n
\end{aligned}
\tag{M1}
\end{equation}

受…的约束
\begin{equation}
\sum_{a \in A_i} x_{i,a} = 1,
\qquad
\sum_{r \in R_i} z_{i,r} = 1,
\qquad \forall i \in I,
\tag{M2}
\end{equation}
\begin{equation}
z_{i,r} \le v_r,
\qquad \forall (i,r) \in \mathcal{F}^R,
\tag{M3}
\end{equation}
\begin{equation}
x_{i,a} \le h_a^{\mathrm{reg}} + y_a,
\qquad \forall (i,a) \in \mathcal{F}^A,
\tag{M4}
\end{equation}
\begin{equation}
y_a \le h_a^{\mathrm{call}},
\qquad \forall a \in A,
\tag{M5}
\end{equation}
\begin{equation}
s_i \ge t_a^{\mathrm{start}} - M(1-x_{i,a}),
\qquad
s_i \le T^{\mathrm{end}},
\qquad \forall (i,a) \in \mathcal{F}^A,
\tag{M6}
\end{equation}
\begin{equation}
\alpha_{i,i',a} \le u_{i,i'},
\qquad
\beta_{i,i',r} \le u_{i,i'},
\tag{M7}
\end{equation}
对于所有$\{(i,a),(i',a)\}\subseteq \mathcal{F}^A$和$\{(i,r),(i',r)\}\subseteq \mathcal{F}^R$，
\begin{equation}
u_{i,i'} + u_{i',i} \le 1,
\qquad \forall \{i,i'\} \subseteq I,
\tag{M8}
\end{equation}
\begin{equation}
u_{i,i''} \ge u_{i,i'} + u_{i',i''} - 1,
\qquad \forall \{i,i',i''\} \subseteq I,
\tag{M9}
\end{equation}
\begin{equation}
\alpha_{i,i',a} + \alpha_{i',i,a} \le x_{i,a},
\qquad
\alpha_{i,i',a} + \alpha_{i',i,a} \le x_{i',a},
\tag{M10}
\end{equation}
\begin{equation}
\alpha_{i,i',a} + \alpha_{i',i,a} \ge x_{i,a} + x_{i',a} - 1,
\tag{M11}
\end{equation}
对于所有$\{(i,a),(i',a)\}\subseteq \mathcal{F}^A$，
\begin{equation}
\beta_{i,i',r} + \beta_{i',i,r} \le z_{i,r},
\qquad
\beta_{i,i',r} + \beta_{i',i,r} \le z_{i',r},
\tag{M12}
\end{equation}
\begin{equation}
\beta_{i,i',r} + \beta_{i',i,r} \ge z_{i,r} + z_{i',r} - 1,
\tag{M13}
\end{equation}
对于所有$\{(i,r),(i',r)\}\subseteq \mathcal{F}^R$，
\begin{equation}
\alpha_{i,i',a} \ge x_{i,a} + x_{i',a} + \beta_{i,i',r} - 2,
\tag{M14}
\end{equation}
\begin{equation}
\beta_{i,i',r} \ge z_{i,r} + z_{i',r} + \alpha_{i,i',a} - 2,
\tag{M15}
\end{equation}
对于所有可行的组合$\{(i,a),(i',a)\}\subseteq \mathcal{F}^A$与$\{(i,r),(i',r)\}\subseteq \mathcal{F}^R$，
\begin{equation}
\eta_n \ge
\sum_{a \in A} \left(c_a^{\mathrm g} g_a^n + c_a^{\mathrm o} o_a^n\right)
+ \sum_{r \in R} \left(c_r^{\mathrm g} g_r^n + c_r^{\mathrm o} o_r^n\right)
+ \sum_{i \in I} c_i^{\mathrm w} w_i^n
- \tau,
\qquad \forall n=1,\dots,N,
\tag{M16}
\end{equation}
\begin{equation}
\eta_n \ge 0,
\qquad \forall n=1,\dots,N,
\tag{M17}
\end{equation}
\begin{equation}
q_{i'}^n \ge q_i^n + d_i^n - M_{\mathrm{seq}}(1-u_{i,i'}),
\qquad
\forall \{i,i'\}\subseteq I,\ i\neq i',\ n=1,\dots,N,
\tag{M18}
\end{equation}
\begin{equation}
q_i^n \ge s_i,
\qquad \forall i \in I,\ n=1,\dots,N,
\tag{M19}
\end{equation}
\begin{equation}
o_a^n \ge q_i^n + d_i^n - t_a^{\mathrm{end}} - M_{\mathrm{anes}}(1-x_{i,a}+y_a),
\qquad
\forall (i,a)\in \mathcal{F}^A,\ n=1,\dots,N,
\tag{M20}
\end{equation}
\begin{equation}
o_r^n \ge q_i^n + d_i^n - T^{\mathrm{end}} - M_{\mathrm{room}}(1-z_{i,r}),
\qquad
\forall (i,r)\in \mathcal{F}^R,\ n=1,\dots,N,
\tag{M21}
\end{equation}
\begin{equation}
w_i^n \ge q_i^n - s_i,
\qquad \forall i \in I,\ n=1,\dots,N,
\tag{M22}
\end{equation}
\begin{equation}
g_a^n \ge
\left(t_a^{\mathrm{end}} - t_a^{\mathrm{start}} - \sum_{i \in I_a} d_i^n x_{i,a}\right) h_a^{\mathrm{reg}}
+ o_a^n,
\qquad \forall a \in A,\ n=1,\dots,N,
\tag{M23}
\end{equation}
\begin{equation}
g_r^n \ge
T^{\mathrm{end}} v_r - \sum_{i \in I_r} d_i^n z_{i,r} + o_r^n,
\qquad \forall r \in R,\ n=1,\dots,N,
\tag{M24}
\end{equation}
\begin{equation}
x_{i,a}, y_a, z_{i,r}, u_{i,i'}, v_r, \alpha_{i,i',a}, \beta_{i,i',r} \in \{0,1\},
\qquad \forall i \in I,\ a \in A,\ r \in R,
\tag{M25}
\end{equation}
\begin{equation}
s_i \ge 0,
\quad
q_i^n, o_a^n, o_r^n, w_i^n, g_a^n, g_r^n \ge 0,
\qquad \forall i \in I,\ a \in A,\ r \in R,\ n=1,\dots,N.
\tag{M26}
\end{equation}
