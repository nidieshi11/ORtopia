\section*{符号与数据}
设$V$为顾客的集合，$\mathcal{R}$为可行的路线集合。对于每一个顾客$i\in V$和每条路线$r\in\mathcal{R}$，设$a_{ir}\in\{0,1\}$为固定路线-顾客关联系数，其中$a_{ir}=1$表示路线$r$服务于顾客$i$，否则为$a_{ir}=0$。设$c_r\in\mathbb{R}$表示路线$r$的固定成本。

\section*{决策变量}
对于每一个$r\in\mathcal{R}$，让
\[
\lambda_r\in\{0,1\}
\]
指示是否选择路线$r$。

\section*{模型}
\begin{align}
\min_{\lambda}\quad
&\sum_{r\in\mathcal{R}} c_r\lambda_r \\
\text{s.t.}\quad
&\sum_{r\in\mathcal{R}} a_{ir}\lambda_r=1,
&&\forall i\in V,\\
&\lambda_r\in\{0,1\},
&&\forall r\in\mathcal{R}.
\end{align}
