\section*{AMR与拣货员协同拣选M0集成模型}

\paragraph{来源} PDF第5--9页，§3--4，M0式(4.1)--(4.42)。保留直接集成主模型，不保留SRMP及另一套位置指派重构。
\subsection*{集合与参数}
$I,R,P,S,U$ 分别为订单、机器人、拣货员、货架和SKU集合；
$o(s),e(s)$ 为各路线的虚拟起终点，$o(r),e(r)$ 为货架上机器人服务序列的虚拟端点。
这些符号是索引标签。所有路径弧不含自环。
$q^I$ 为每机器人订单数上限；$n^O_{iu}$ 为订单需求，$n^S_{su}$ 为货架库存。
$w^U_u,v^U_u$ 为SKU单位重量/体积；$w^R,v^R$ 为机器人承载上限。
$t^R_{ss^{\prime}},t^P_{ss^{\prime}}$ 为机器人/拣货员货架间（含结束点）行走时间，
$t^{R0}_{rs},t^{P0}_{ps}$ 为从各自当前位置至首站的时间。
$h$ 为每个被拣取SKU种类的处理时间：本文(4.27)、(4.29)、(4.31)明确使用
$h\sum_u\mu_{rsu}$，按所列货架操作指示量计时。
$\dot M,\ddot M,M^{27}_{ss^{\prime}},M^{29}_{ss^{\prime}},M^{30}_s,M^{31}$ 为足够大的有效上界。
\subsection*{变量及含义}
$\alpha$ 为订单指派，$\beta,\gamma$ 为机器人/拣货员访问货架；
$\#_{rsu}$ 为取货数量，$\mu$ 表示该SKU是否被取；
$\delta,\varepsilon$ 是机器人、拣货员路径，$\sigma$ 是同货架机器人服务先后弧；
带波浪符号为对应整数顺序；$\epsilon,\theta$ 为到达时间，$\rho$ 为开始拣货时间。

\subsection*{唯一目标和全部约束}
\[\min\ \max_{r\in R}\epsilon_{r,e(s)}.\tag{4.1}\]
\[\begin{aligned}&\text{订单分配与机器人路径:}\end{aligned}\]
\[\begin{aligned}& \sum_{r \in R} \alpha_{i, r} = 1 \quad \forall i \in I\end{aligned}\]
\[\begin{aligned}& \sum_{i \in I} \alpha_{i, r} \le q^I \quad \forall r \in R\end{aligned}\]
\[\begin{aligned}& \sum_{p \in P} \gamma_{p, s} \le 1 \quad \forall s \in S\end{aligned}\]
\[\begin{aligned}& \sum_{p \in P} \gamma_{p, s} \le \sum_{r \in R} \beta_{r, s} \quad \forall s \in S\end{aligned}\]
\[\begin{aligned}& \beta_{r, s} \le \sum_{p \in P} \gamma_{p, s} \quad \forall r \in R, s \in S\end{aligned}\]
\[\begin{aligned}& \mu_{r, s, u} \le \#_{r, s, u} \le n^S_{s, u} \mu_{r, s, u} \quad \forall r \in R, s \in S, u \in U\end{aligned}\]
\[\begin{aligned}& \#_{r, s, u} \le n^S_{s, u} \beta_{r, s} \quad \forall r \in R, s \in S, u \in U\end{aligned}\]
\[\begin{aligned}& \beta_{r, s} \le \sum_{u \in U} \mu_{r, s, u} \quad \forall r \in R, s \in S\end{aligned}\]
\[\begin{aligned}& \sum_{i \in I} n^O_{i, u} \alpha_{i, r} = \sum_{s \in S} \#_{r, s, u} \quad \forall r \in R, u \in U\end{aligned}\]
\[\begin{aligned}& \sum_{r \in R} \#_{r, s, u} \le n^S_{s, u} \quad \forall s \in S, u \in U\end{aligned}\]
\[\begin{aligned}& \sum_{s \in S} \sum_{u \in U} w^U_u \#_{r, s, u} \le w^R \quad \forall r \in R\end{aligned}\]
\[\begin{aligned}& \sum_{s \in S} \sum_{u \in U} v^U_u \#_{r, s, u} \le v^R \quad \forall r \in R\end{aligned}\]
\[\begin{aligned}& \sum_{s \in S \cup \{o(s)\}} \delta_{r, s, s^{\prime}} = \sum_{s \in S \cup \{e(s)\}} \delta_{r, s^{\prime}, s} = \beta_{r, s^{\prime}} \quad \forall r \in R, s^{\prime} \in S\end{aligned}\]
\[\begin{aligned}& \sum_{s \in S \cup \{o(s)\}} \delta_{r, s, e(s)} = \sum_{s \in S \cup \{e(s)\}} \delta_{r, o(s), s} = 1 \quad \forall r \in R\end{aligned}\]
\[\begin{aligned}& \tilde{\delta}_{r, s} + 1 - \dot{M}(1 - \delta_{r, s, s^{\prime}}) \le \tilde{\delta}_{r, s^{\prime}} \quad \forall r \in R, s \in S \cup \{o(s)\}, s^{\prime} \in S \cup \{e(s)\}\end{aligned}\]
\[\begin{aligned}& \tilde{\delta}_{r, s} \le \sum_{s^{\prime} \in S} \beta_{r, s^{\prime}} \quad \forall r \in R, s \in S\end{aligned}\]
\[\begin{aligned}& \sum_{s \in S \cup \{o(s)\}} \varepsilon_{p, s, s^{\prime}} = \sum_{s \in S \cup \{e(s)\}} \varepsilon_{p, s^{\prime}, s} = \gamma_{p, s^{\prime}} \quad \forall p \in P, s^{\prime} \in S\end{aligned}\]
\[\begin{aligned}& \sum_{s \in S \cup \{o(s)\}} \varepsilon_{p, s, e(s)} = \sum_{s \in S \cup \{e(s)\}} \varepsilon_{p, o(s), s} = 1 \quad \forall p \in P\end{aligned}\]
\[\begin{aligned}& \tilde{\varepsilon}_{p, s} + 1 - \dot{M}(1 - \varepsilon_{p, s, s^{\prime}}) \le \tilde{\varepsilon}_{p, s^{\prime}} \quad \forall p \in P, s \in S \cup \{o(s)\}, s^{\prime} \in S \cup \{e(s)\}\end{aligned}\]
\[\begin{aligned}& \tilde{\varepsilon}_{p, s} \le \sum_{s^{\prime} \in S} \gamma_{p, s^{\prime}} \quad \forall p \in P, s \in S\end{aligned}\]
\[\begin{aligned}& \sum_{r \in R \cup \{o(r)\}} \sigma_{s, r, r^{\prime}} = \sum_{r \in R \cup \{e(r)\}} \sigma_{s, r^{\prime}, r} = \beta_{r^{\prime}, s} \quad \forall s \in S, r^{\prime} \in R\end{aligned}\]
\[\begin{aligned}& \sum_{r \in R \cup \{o(r)\}} \sigma_{s, r, e(r)} = \sum_{r \in R \cup \{e(r)\}} \sigma_{s, o(r), r} = 1 \quad \forall s \in S\end{aligned}\]
\[\begin{aligned}& \tilde{\sigma}_{s, r} + 1 - \ddot{M}(1 - \sigma_{s, r, r^{\prime}}) \le \tilde{\sigma}_{s, r^{\prime}} \quad \forall s \in S, r \in R \cup \{o(r)\}, r^{\prime} \in R \cup \{e(r)\}\end{aligned}\]
\[\begin{aligned}& \tilde{\sigma}_{s, r} \le \sum_{r^{\prime} \in R} \beta_{r^{\prime}, s} \quad \forall r \in R, s \in S\end{aligned}\]
\[\begin{aligned}& \sum_{s \in S} s \gamma_{p, s} \le \sum_{s \in S} s \gamma_{p^{\prime}, s} \quad \forall p, p^{\prime} \in P, p < p^{\prime}\end{aligned}\]
\[\begin{aligned}& \sum_{i \in I} i \alpha_{i, r} \le \sum_{i \in I} i \alpha_{i, r^{\prime}} \quad \forall r, r^{\prime} \in R, r < r^{\prime}\end{aligned}\]\[\begin{aligned}&\text{时间约束:}\end{aligned}\]
\[\begin{aligned}& \epsilon_{r, s} \ge t^{R0}_{r, s} \delta_{r, o(s), s} \quad \forall r \in R, s \in S \cup \{e(s)\}\end{aligned}\]
\[\begin{aligned}& \epsilon_{r, s^{\prime}} \ge \rho_{r, s} + h \sum_{u \in U} \mu_{r, s, u} + t^R_{s, s^{\prime}} - \overset{\cdots}{M}^{27}_{s, s^{\prime}}(1 - \delta_{r, s, s^{\prime}}) \quad \forall r \in R, s \in S, s^{\prime} \in S \cup \{e(s)\}\end{aligned}\]
\[\begin{aligned}& \theta_{p, s} \ge t^{P0}_{p, s} \varepsilon_{p, o(s), s} \quad \forall p \in P, s \in S \cup \{e(s)\}\end{aligned}\]
\[\begin{aligned}& \theta_{p, s^{\prime}} \ge \rho_{r, s} + h \sum_{u \in U} \mu_{r, s, u} + t^P_{s, s^{\prime}} - \overset{\cdots}{M}^{29}_{s, s^{\prime}}(1 - \varepsilon_{p, s, s^{\prime}}) \quad \forall r \in R, p \in P, s \in S, s^{\prime} \in S \cup \{e(s)\}\end{aligned}\]
\[\begin{aligned}& \rho_{r, s} \ge \max\{\epsilon_{r, s}, \theta_{p, s}\} - \overset{\cdots}{M}^{30}_s (2 - \beta_{r, s} - \gamma_{p, s}) \quad \forall r \in R, s \in S, p \in P\end{aligned}\]
\[\begin{aligned}& \rho_{r^{\prime}, s} \ge \rho_{r, s} + h \sum_{u \in U} \mu_{r, s, u} - \overset{\cdots}{M}^{31} (1 - \sigma_{s, r, r^{\prime}}) \quad \forall r, r^{\prime} \in R, s \in S\end{aligned}\]
\[\begin{aligned}& \epsilon_{r, s} \le \overset{\cdots}{M}^{30}_s \beta_{r, s} \quad \forall r \in R, s \in S\end{aligned}\]
\[\begin{aligned}& \theta_{p, s} \le \overset{\cdots}{M}^{30}_s \gamma_{p, s} \quad \forall p \in P, s \in S\end{aligned}\]
\[\begin{aligned}& \rho_{r, s} \le \overset{\cdots}{M}^{30}_s \beta_{r, s} \quad \forall r \in R, s \in S\end{aligned}\]
路线起点的顺序值固定为零，虚拟终点允许取“实际访问数加一”；
空路线为虚拟起点直接到终点，实际货架未访问时相关到达及服务时刻由上式固定为零。
所有数量域按原文(4.38)为连续非负，$\mathbb Z^+$ 在本整理中表示非负整数；
并不自行加上原文没有的取货整数限制。
