\section{集合与数据}
让$T=\{1,\ldots,H\}$保持不变，其中$H\geq1$表示周期的数量。让$G$、$I$、$B$、$L$、$W$、$P$和$D$分别表示有限个的热发生器、插值顶点、母线、传输线路、风力发电单元、储能单元以及负载。对于每一个母线$b\in B$，给定的关联集合$G_b\subseteq G$、$W_b\subseteq W$、$P_b\subseteq P$和$D_b\subseteq D$分别标识了位于该母线上的设备和负载。每条线路$l\in L$都有起点母线$sb_l\in B$和终点母线$eb_l\in B$；让$L_b^{\mathrm{in}}=\{l\in L:eb_l=b\}$和$L_b^{\mathrm{out}}=\{l\in L:sb_l=b\}$分别表示起点母线和终点母线。此外，还有一个参考母线$\mathrm{ref}\in B$。

以下所有参数都是固定大小的标量数据，其位置与原始顺序相同。请保留所有的数字符号。
\begin{itemize}
\item $\Delta t>0$：周期持续时间；$c_{gi}$和$p_{gi}$、$(g,i)\in G\times I$：插值顶点处的生成成本与生成输出值。 
\item $C_p^{ls}$和$C_p^{gs}$：分别表示负载削减和电网削减的惩罚系数。下标$p$用于标识这些惩罚系数。
\item $UT_g,DT_g\in\mathbb Z_{\geq1}$：最小上下波动时间；$P_g^{\min},P_g^{\max}$：生成限制；$R_g^{up},R_g^{down},R_g^{SU},R_g^{SD}$：针对$g\in G$的逐步增加、逐步减少、启动阶段以及关闭阶段的限制。
\item $I_w^t$：可用于$(w,t)\in W\times T$的风力发电能力；$d_d^t$：用于$(d,t)\in D\times T$的需求量。
\item $\eta_p^{chg}$和$\eta_p^{disc}>0$：充放电效率；$Cap_p^{chg},Cap_p^{disc}$：充放电功率容量；$S_p^{\min},S_p^{\max}$：存储状态的限制，适用于$p\in P$。
\item $B_l$: 线路电导；$F_l$: 线路流量容量，适用于$l\in L$。
\item $x_g^0,p_g^0,s_p^0$以及所有启动和关闭历史记录值$u_g^\tau,v_g^\tau$，其中$\tau<1$出现在约束条件中：已知的初始条件数据。
\end{itemize}
在以下$1$时间点引用的所有数量均为固定的初始条件。

\section{决策变量}
对于每一个被建模的期间$t\in T$，相关的变量如下：
\begin{itemize}
\item $x_g^t\in\{0,1\}$：承诺状态，而$u_g^t,v_g^t\in\mathbb R_{\geq0}$：启动和关闭变量，适用于$g\in G$。
\item $w_{gi}^t\in\mathbb R_{\geq0}$：插值权重，用于$(g,i)\in G\times I$；$p_g^t\in\mathbb R$：热发电，用于$g\in G$。
\item $g_w^t,c_w^t\in\mathbb R_{\geq0}$: 风电发电与电力削峰措施，适用于$w\in W$。
\item $pc_p^t,pd_p^t,s_p^t\in\mathbb R$: 用于$p\in P$的充电功率、放电功率以及储存的能量。
\item $f_l^t\in\mathbb R$：线路流量，用于$l\in L$；$\theta_b^t\in\mathbb R$：电压角度，用于$b\in B$。
\item $ls_b^t,gs_b^t\in\mathbb R_{\geq0}$，对于$b\in B$而言：指的是非负的负荷削减以及多余的发电量，具体如源表3中所定义的那样。
\end{itemize}
连续变量遵循所列出的领域。风力发电、削减和弃风都是非负物理量。源方程（5m）中，第7页的$gs_b^t$被用于节点平衡计算；提取出的独立变量$ps_b^t$是抄写错误，因此被删除了。

\section{优化模型}
有限时间范围内的目标，结合了承诺方程和全时间范围内的运营成本。目标是将这些因素进行最小化。
\begin{equation}
\min\quad
\sum_{t\in T}\left[
\sum_{g\in G}\sum_{i\in I}c_{gi}w_{gi}^t
+\sum_{b\in B}\left(C_p^{ls}ls_b^t+C_p^{gs}gs_b^t\right)
\right]\Delta t
\end{equation}
遵循以下限制条件。

\subsection*{承诺/保证}
\begin{align}
x_g^t-x_g^{t-1}&=u_g^t-v_g^t
&&g\in G,\ t\in T,\\
\sum_{k=0}^{UT_g-1}u_g^{t-k}&\leq x_g^t
&&g\in G,\ t\in T,\\
\sum_{k=0}^{DT_g-1}v_g^{t-k}&\leq1-x_g^t
&&g\in G,\ t\in T.
\end{align}

\subsection*{热生成}
\begin{align}
p_g^t&=\sum_{i\in I}w_{gi}^t p_{gi}
&&g\in G,\ t\in T,\\
\sum_{i\in I}w_{gi}^t&=x_g^t
&&g\in G,\ t\in T,\\
x_g^tP_g^{\min}\leq p_g^t&\leq x_g^tP_g^{\max}
&&g\in G,\ t\in T,\\
p_g^t-p_g^{t-1}&\leq x_g^{t-1}R_g^{up}+(1-x_g^{t-1})R_g^{SU}
&&g\in G,\ t\in T,\\
p_g^{t-1}-p_g^t&\leq x_g^tR_g^{down}+(1-x_g^t)R_g^{SD}
&&g\in G,\ t\in T.
\end{align}

\subsection*{风与储存}
\begin{align}
g_w^t+c_w^t&=I_w^t
&&w\in W,\ t\in T,\\
s_p^t&=s_p^{t-1}+\left(\eta_p^{chg}pc_p^t-\frac{pd_p^t}{\eta_p^{disc}}\right)\Delta t
&&p\in P,\ t\in T,\\
0\leq pc_p^t&\leq Cap_p^{chg}
&&p\in P,\ t\in T,\\
0\leq pd_p^t&\leq Cap_p^{disc}
&&p\in P,\ t\in T,\\
S_p^{\min}\leq s_p^t&\leq S_p^{\max}
&&p\in P,\ t\in T.
\end{align}

\subsection*{网络与边界的消除}
\begin{align}
f_l^t&=B_l\left(\theta_{eb_l}^t-\theta_{sb_l}^t\right)
&&l\in L,\ t\in T,\\
\theta_{\mathrm{ref}}^t&=0
&&t\in T,\\
-F_l\leq f_l^t&\leq F_l
&&l\in L,\ t\in T,\\
ls_b^t&\leq\sum_{d\in D_b}d_d^t
&&b\in B,\ t\in T,\\
gs_b^t&\leq\sum_{g\in G_b}p_g^t
&&b\in B,\ t\in T.
\end{align}
节点平衡是
\begin{equation}
\begin{aligned}
&\sum_{g\in G_b}p_g^t+\sum_{w\in W_b}g_w^t
+\sum_{p\in P_b}pd_p^t+\sum_{l\in L_b^{\mathrm{in}}}f_l^t+ls_b^t\\
&\qquad=\sum_{d\in D_b}d_d^t
+\sum_{l\in L_b^{\mathrm{out}}}f_l^t
+\sum_{p\in P_b}pc_p^t+gs_b^t,
\qquad b\in B,\ t\in T.
\end{aligned}
\end{equation}
