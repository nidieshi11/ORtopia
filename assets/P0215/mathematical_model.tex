\section*{选择与路由模型}

设$\mathcal M$、$\Pi$、$\mathcal L$和$\mathcal Q$分别为地雷、港口、粒度属性和属性的有限集合。规划周期为$1,\ldots,T$。地雷$m$具有$N_m$个固定的候选调度计划$\vec s_{m,j}$，这些计划由$j=1,\ldots,N_m$索引；$N_{\pi,l}$是港口$\pi$处粒度属性$l$的产品集合。给定的整数值路由窗口端点满足$1\leq f(t)\leq t$和$t'\leq g(t')\leq T$条件。

标量参数包括$\beta_1,\beta_2,T_R$，候选值$P_j^m$，端口容量$C_\pi^t$，产品质量上限$L_{\pi,l,n,q}^{t}$和$U_{\pi,l,n,q}^{t}$，以及属性偏差系数$\lambda_q^{+}$。对于每个固定的候选调度方案，$\tau_l^m(\vec s_{m,j},t')$表示其生产量，而$v_{l,q}^{m,t'}(\vec s_{m,j})$则表示其组成值。 

定义路由索引集合
\[
\mathcal W=\{(t,t'):1\leq t'\leq t\leq T,\ f(t)\leq t'\}
\cup\{(t,t'):1\leq t'\leq t\leq T,\ t\leq g(t')\}.
\]
变量包括二进制调度选择$s_{m,j}$、非负路由量$r_{\pi,n,t}^{m,l,j,t'}$（用于$(t,t')\in\mathcal W$）、不受限制的仿射表达式变量$z_{\pi,l,n,q}^{+,t},z_{\pi,l,n,q}^{-,t}$，以及非负正部分变量$d_{\pi,l,n,q}^{+,t},d_{\pi,l,n,q}^{-,t}$。变量$z$和$d$的索引分别由$\pi\in\Pi$、$l\in\mathcal L$、$n\in N_{\pi,l}$、$q\in\mathcal Q$和$t=1,\ldots,T$确定。

\begin{align}
\min\quad &\sum_{t=1}^{T}\left[
\beta_1\sum_{\pi\in\Pi}\sum_{l\in\mathcal L}\sum_{n\in N_{\pi,l}}\sum_{q\in\mathcal Q}
T_R\lambda_q^{+}\bigl(d_{\pi,l,n,q}^{+,t}+d_{\pi,l,n,q}^{-,t}\bigr)
-\beta_2\sum_{m\in\mathcal M}\sum_{j=1}^{N_m}s_{m,j}P_j^m
\right] \
\text{subject to}\quad
&\sum_{j=1}^{N_m}s_{m,j}=1
&&\forall m\in\mathcal M,\\
&\sum_{l\in\mathcal L}\sum_{n\in N_{\pi,l}}\sum_{m\in\mathcal M}
\sum_{j=1}^{N_m}\sum_{t'=f(t)}^{t}T_Rr_{\pi,n,t}^{m,l,j,t'}
\leq C_\pi^t
&&\forall\pi\in\Pi,\ t=1,\ldots,T,\\
&\sum_{\pi\in\Pi}\sum_{n\in N_{\pi,l}}
\sum_{t=t'}^{g(t')}T_Rr_{\pi,n,t}^{m,l,j,t'}
=s_{m,j}\tau_l^m(\vec s_{m,j},t')
&&\forall m\in\mathcal M,\ j=1,\ldots,N_m,\ l\in\mathcal L,\ t'=1,\ldots,T,\\
&z_{\pi,l,n,q}^{+,t}=
\sum_{m\in\mathcal M}\sum_{j=1}^{N_m}\sum_{t'=f(t)}^{t}
 r_{\pi,n,t}^{m,l,j,t'}v_{l,q}^{m,t'}(\vec s_{m,j})
-U_{\pi,l,n,q}^{t}
\sum_{m\in\mathcal M}\sum_{j=1}^{N_m}\sum_{t'=f(t)}^{t}r_{\pi,n,t}^{m,l,j,t'}
&&\forall\pi,l,n,q,t,\\
&z_{\pi,l,n,q}^{-,t}=
L_{\pi,l,n,q}^{t}
\sum_{m\in\mathcal M}\sum_{j=1}^{N_m}\sum_{t'=f(t)}^{t}r_{\pi,n,t}^{m,l,j,t'}
-\sum_{m\in\mathcal M}\sum_{j=1}^{N_m}\sum_{t'=f(t)}^{t}
 r_{\pi,n,t}^{m,l,j,t'}v_{l,q}^{m,t'}(\vec s_{m,j})
&&\forall\pi,l,n,q,t,\\
&d_{\pi,l,n,q}^{+,t}=\max\{0,z_{\pi,l,n,q}^{+,t}\},\qquad
 d_{\pi,l,n,q}^{-,t}=\max\{0,z_{\pi,l,n,q}^{-,t}\}
&&\forall\pi,l,n,q,t,\\
&s_{m,j}\in\{0,1\}
&&\forall m\in\mathcal M,\ j=1,\ldots,N_m,\\
&r_{\pi,n,t}^{m,l,j,t'}\geq0
&&\forall\pi\in\Pi,\ l\in\mathcal L,\ n\in N_{\pi,l},\ m\in\mathcal M,\ j=1,\ldots,N_m,\ (t,t')\in\mathcal W,\\
&z_{\pi,l,n,q}^{+,t},z_{\pi,l,n,q}^{-,t}\in\mathbb R,\qquad
 d_{\pi,l,n,q}^{+,t},d_{\pi,l,n,q}^{-,t}\geq0
&&\forall\pi,l,n,q,t.
\end{align}
在此处及上方，$\forall\pi,l,n,q,t$分别简称为$\pi\in\Pi$、$l\in\mathcal L$、$n\in N_{\pi,l}$、$q\in\mathcal Q$和$t=1,\ldots,T$。所有数字保持不变。
