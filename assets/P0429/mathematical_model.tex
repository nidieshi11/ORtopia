\section*{集合与索引}
设\(\mathcal N\)为一个有限的节点集，\(\mathcal E\subseteq\mathcal N\times\mathcal N\)为一个有限的有向边集。设\(\mathcal E_h\subseteq\mathcal E\)为候选边的集合，\(\mathcal N_h\subseteq\mathcal N\)为候选节点的集合。对于每一个\(i\in\mathcal N_h\)，设\(\mathcal E_h^{-}(i)\subseteq\mathcal E_h\)和\(\mathcal E_h^{+}(i)\subseteq\mathcal E_h\)分别表示指定的进入候选边的集合和离开候选边的集合，并定义相关的候选边集合。
\[
\mathcal E_h(i):=\mathcal E_h^{-}(i)\cup\mathcal E_h^{+}(i).
\]
让\(\mathcal S\subseteq\mathcal N\times\mathcal N\)表示起点与终点对的有限集合，这些对的索引由\((o,d)\)表示。让\(\mathcal R\)表示流量平衡方程的有限行索引集合。

\section*{参数}
\begin{itemize}
    \item \(q^{od}\in\mathbb R_+\)：OD对的重量，对应\((o,d)\in\mathcal S\)。
    \item \(\mathbf c=(c_{ij})_{(i,j)\in\mathcal E_h}\in\mathbb R^{|\mathcal E_h|}\): 候选边成本向量。
    \item \(\mathbf b=(b_i)_{i\in\mathcal N_h}\in\mathbb R^{|\mathcal N_h|}\): 候选节点成本向量。
    \item \(B_{\mathrm{edge}},B_{\mathrm{node}}\in\mathbb R\)：边和节点的预算。
    \item \(\mathbf A\in\mathbb R^{|\mathcal R|\times|\mathcal E|}\): 流量平衡矩阵，其列索引由\(\mathcal E\)表示。
    \item \(\mathbf e^{od}\in\mathbb R^{|\mathcal R|}\): 每个\((o,d)\in\mathcal S\)的右侧流量平衡。
    \item \(\mathbf t=(t_{ij})_{(i,j)\in\mathcal E}\in\mathbb R^{|\mathcal E|}\)：边沿行进时间向量。
    \item \(\beta_1,\beta_2\in\mathbb R\)：满足\(\beta_1\leq\beta_2\)的仿射片段斜率系数。
    \item \(T_1\in\mathbb R\): 用于定义仿射片段截距的断点
    \[
    \alpha_1:=1,
    \qquad
    \alpha_2:=1+(\beta_2-\beta_1)T_1.
    \]
\end{itemize}

\section*{决策变量}
对于每一个\((o,d)\in\mathcal S\)，让
\[
\mathbf y^{od}=(y_{ij}^{od})_{(i,j)\in\mathcal E}\in\mathbb R^{|\mathcal E|}
\]
设为路由流向量，令\(v^{od}\in\mathbb R\)为其分段线性性能值。对于每一个\((i,j)\in\mathcal E_h\)，令\(x_{ij}\)为候选边设计变量；对于每一个\(i\in\mathcal N_h\)，令\(z_i\)为候选节点设计变量。

\section*{混合整数线性规划}
\begin{align}
\max_{\mathbf v,\mathbf x,\mathbf y,\mathbf z}\quad
& \sum_{(o,d)\in\mathcal S}q^{od}v^{od} \label{obj}\\
\text{subject to}\quad
& \sum_{(i,j)\in\mathcal E_h}c_{ij}x_{ij}
  \leq B_{\mathrm{edge}}, \label{edge-budget}\\
& \sum_{i\in\mathcal N_h}b_i z_i
  \leq B_{\mathrm{node}}, \label{node-budget}\\
& \mathbf A\mathbf y^{od}=\mathbf e^{od},
  && \forall (o,d)\in\mathcal S, \label{flow-balance}\\
& y_{ij}^{od}\leq x_{ij},
  && \forall (i,j)\in\mathcal E_h,\ \forall (o,d)\in\mathcal S, \label{edge-linking}\\
& y_{ij}^{od}\leq x_{wl}+z_i,
  && \forall i\in\mathcal N_h,\ \forall (i,j)\in\mathcal E_h^{-}(i),
     \ \forall (w,l)\in\mathcal E_h(i),\ \forall (o,d)\in\mathcal S, \label{node-linking}\\
& v^{od}\leq \alpha_1-\beta_1\sum_{(i,j)\in\mathcal E}t_{ij}y_{ij}^{od},
  && \forall (o,d)\in\mathcal S, \label{piece-one}\\
& v^{od}\leq \alpha_2-\beta_2\sum_{(i,j)\in\mathcal E}t_{ij}y_{ij}^{od},
  && \forall (o,d)\in\mathcal S, \label{piece-two}\\
& v^{od}\geq 0,
  && \forall (o,d)\in\mathcal S, \label{value-domain}\\
& 0\leq y_{ij}^{od}\leq 1,
  && \forall (i,j)\in\mathcal E,\ \forall (o,d)\in\mathcal S, \label{flow-domain}\\
& x_{ij}\in\{0,1\},
  && \forall (i,j)\in\mathcal E_h, \label{edge-domain}\\
& z_i\in\{0,1\},
  && \forall i\in\mathcal N_h. \label{node-domain}
\end{align}
