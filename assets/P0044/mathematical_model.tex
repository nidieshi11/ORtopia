\section{Exact MBLP Reformulation}

\begin{equation}
\min \quad \boldsymbol{b}^{\top}\boldsymbol{\beta} + \sum_{k\in\mathcal{K}} \boldsymbol{b}^{\top}\boldsymbol{\beta}^k
\tag{EC.12}
\end{equation}

\begin{equation}
\begin{aligned}
\text{s.t.}\quad
& \boldsymbol{x}\in\mathcal{X},\ \boldsymbol{w}\in\mathcal{W},\ \boldsymbol{y}^k\in\mathcal{Y}
&& \forall k\in\mathcal{K}, \\
& \boldsymbol{\alpha}\in\mathbb{R}_+^{K},\ \boldsymbol{\beta}\in\mathbb{R}_+^{R},\ \boldsymbol{\beta}^k\in\mathbb{R}_+^{R},\ \boldsymbol{\gamma}^k\in\mathbb{R}^{N_\xi}
&& \forall k\in\mathcal{K}, \\
& \bar{\boldsymbol{\gamma}}^k\in\mathbb{R}^{N_\xi},\ \bar{\boldsymbol{x}}^k\in\mathbb{R}_+^{N_x},\ \bar{\boldsymbol{w}}^k\in\mathbb{R}_+^{N_\xi},\ \bar{\boldsymbol{y}}^k\in\mathbb{R}_+^{N_y}
&& \forall k\in\mathcal{K}, \\
& \boldsymbol{e}^{\top}\boldsymbol{\alpha}=1, \\
& \boldsymbol{A}^{\top}\boldsymbol{\beta}=\sum_{k\in\mathcal{K}}\bar{\boldsymbol{\gamma}}^k, \\
& \boldsymbol{A}^{\top}\boldsymbol{\beta}^k+\bar{\boldsymbol{\gamma}}^k
= \boldsymbol{C}\bar{\boldsymbol{x}}^k+\boldsymbol{D}\bar{\boldsymbol{w}}^k+\boldsymbol{Q}\bar{\boldsymbol{y}}^k
&& \forall k\in\mathcal{K}, \\
& \boldsymbol{T}\boldsymbol{x}+\boldsymbol{V}\boldsymbol{w}+\boldsymbol{W}\boldsymbol{y}^k \le \boldsymbol{h}
&& \forall k\in\mathcal{K}, \\
& \bar{\boldsymbol{x}}^k \le \boldsymbol{x},\quad
\bar{\boldsymbol{x}}^k \le \alpha_k \boldsymbol{e},\quad
\bar{\boldsymbol{x}}^k \ge (\alpha_k-1)\boldsymbol{e}+\boldsymbol{x}
&& \forall k\in\mathcal{K}, \\
& \bar{\boldsymbol{w}}^k \le \boldsymbol{w},\quad
\bar{\boldsymbol{w}}^k \le \alpha_k \boldsymbol{e},\quad
\bar{\boldsymbol{w}}^k \ge (\alpha_k-1)\boldsymbol{e}+\boldsymbol{w}
&& \forall k\in\mathcal{K}, \\
& \bar{\boldsymbol{y}}^k \le \boldsymbol{y}^k,\quad
\bar{\boldsymbol{y}}^k \le \alpha_k \boldsymbol{e},\quad
\bar{\boldsymbol{y}}^k \ge (\alpha_k-1)\boldsymbol{e}+\boldsymbol{y}^k
&& \forall k\in\mathcal{K}, \\
& \bar{\boldsymbol{\gamma}}^k \le \boldsymbol{\gamma}^k + M(\boldsymbol{e}-\boldsymbol{w}),\quad
\bar{\boldsymbol{\gamma}}^k \le M\boldsymbol{w},
&& \forall k\in\mathcal{K}, \\
& \bar{\boldsymbol{\gamma}}^k \ge -M\boldsymbol{w},\quad
\bar{\boldsymbol{\gamma}}^k \ge \boldsymbol{\gamma}^k - M(\boldsymbol{e}-\boldsymbol{w})
&& \forall k\in\mathcal{K}.
\end{aligned}
\end{equation}
