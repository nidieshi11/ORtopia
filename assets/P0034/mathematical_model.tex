\section*{Single-Level MISOCP Reformulation}
The original planning model is bilevel. The retained core model is the single-level MISOCP
reformulation that the paper actually solves.

\begin{equation}
\begin{aligned}
\min \quad
& \sum_{i \in I} c_i^{\mathrm r} y_i
+ \sum_{m \in N} c_m^{\mathrm p} z_m
+ \sum_{(m,n) \in L} c_{mn}^{\mathrm u} u_{mn} \\
& \qquad
+ \xi \tau \sum_{m \in N} \left(c_m^{\mathrm h} p_m^{\mathrm h} + c_m^{\mathrm i} p_m^{\mathrm i}\right)
+ \xi \sum_{m \in N} \sum_{i \in I} c_{mi}^{\mathrm t} f_{mi}
\end{aligned}
\tag{M1}
\end{equation}

subject to
\begin{equation}
q_i \le x_i,
\qquad \forall i \in I,
\tag{M2}
\end{equation}
\begin{equation}
\sum_{r \in R_{(i,j,i)}} l_r w_{(i,j,i)r}
+ \bar{l}_{(i,j,i)} \big(w_{(i,j,i)0} + \phi_{ij} - 1\big)
\le (1+\gamma) d_{ij},
\qquad \forall i,j \in I,\ i \neq j,
\tag{M3}
\end{equation}
\begin{equation}
(1+\gamma) d_{ij}
\le
\sum_{r \in R_{(i,j,i)}} l_r w_{(i,j,i)r}
+ \bar{l}_{(i,j,i)} \big(w_{(i,j,i)0} + \phi_{ij}\big),
\qquad \forall i,j \in I,\ i \neq j,
\tag{M4}
\end{equation}
\begin{equation}
\phi_{ii} = x_i,
\qquad \forall i \in I,
\tag{M5}
\end{equation}
\begin{equation}
b_i u_i \le \sum_{j \in I} \mu_{ij} \phi_{ij},
\qquad \forall i \in I,
\tag{M6}
\end{equation}
\begin{equation}
\sum_{j \in I} \mu_{ij} \phi_{ij}
\le
\sum_{j \in I} \mu_{ij} u_i + b_i,
\qquad \forall i \in I,
\tag{M7}
\end{equation}
\begin{equation}
\left\lVert
\begin{array}{c}
q_i - \beta_i \\
2 \alpha_i
\end{array}
\right\rVert_2
\le q_i + \beta_i,
\qquad \forall i \in I,
\tag{M8}
\end{equation}
\begin{equation}
\left\lVert
\begin{array}{c}
1 - q_i - \beta_i \\
2 \Sigma_i^{1/2} \phi_i
\end{array}
\right\rVert_2
\le 1 - q_i + \beta_i,
\qquad \forall i \in I,
\tag{M9}
\end{equation}
\begin{equation}
\alpha_i = \sum_{j \in I} \mu_{ij} \epsilon_{ijj} - b_i v_i,
\qquad \forall i \in I,
\tag{M10}
\end{equation}
\begin{equation}
\beta_i =
\sum_{j_1 \in I} \sum_{j_2 \in I} \sigma_{(ij_1)(ij_2)} \delta_{ij_1 j_2}
+ \sum_{j_1 \in I} \sum_{j_2 \in I} \mu_{ij_1}\mu_{ij_2} \epsilon_{ij_1 j_2}
+ b_i^2 v_i
- 2 b_i \sum_{j \in I} \mu_{ij} \epsilon_{ijj},
\quad \forall i \in I,
\tag{M11}
\end{equation}
\begin{equation}
(v_i,x_i,u_i) \in \mathcal A,
\qquad
(\delta_{ij_1 j_2},\phi_{ij_1},\phi_{ij_2}) \in \mathcal A,
\qquad
(\epsilon_{ij_1 j_2},v_i,\phi_{ij_1},\phi_{ij_2}) \in \mathcal B,
\tag{M12}
\end{equation}
where
\begin{equation}
\mathcal A =
\left\{
(v,x,u) \in \{0,1\}^3 :
v \le x,\ v \le u,\ v \ge x+u-1
\right\},
\tag{M13}
\end{equation}
\begin{equation}
\mathcal B =
\left\{
(v,x,y,z) \in \{0,1\}^4 :
v \le x,\ v \le y,\ v \le z,\ v \ge x+y+z-2
\right\},
\tag{M14}
\end{equation}
\begin{equation}
D_k = \sum_{t \in T} \sum_{r \in R_t} N_t \eta_{rk} \kappa_{tr},
\qquad \forall k \in I,
\tag{M15}
\end{equation}
\begin{equation}
\kappa_{tr} \ge 0,\qquad
\kappa_{tr} \le q_i,\qquad
\kappa_{tr} \le w_{tr},\qquad
\kappa_{tr} \ge q_i + w_{tr} - 1,
\quad \forall i \in I,\ t \in T_i,\ r \in R_t,
\tag{M16}
\end{equation}
\begin{equation}
D_i \le y_i,
\qquad
y_i \le Y_i x_i,
\qquad \forall i \in I,
\tag{M17}
\end{equation}
\begin{equation}
(1-\Gamma)\sum_{i \in I}\sum_{t:o(t)=i} N_t \sum_{r \in R_t} l_r \kappa_{tr}
\ge
\Gamma \sum_{i \in I} (1-q_i)\sum_{t:o(t)=i} s_t N_t,
\tag{M18}
\end{equation}
\begin{equation}
p_{mn} = b_{mn}(\theta_m - \theta_n),
\qquad \forall (m,n) \in L,
\tag{M19}
\end{equation}
\begin{equation}
\sum_{\substack{n \in N:\\(m,n)\in L}} p_{mn} + p_m^{\mathrm c}
=
\sum_{\substack{n \in N:\\(n,m)\in L}} p_{nm} + p_m^{\mathrm h} + p_m^{\mathrm i},
\qquad \forall m \in N,
\tag{M20}
\end{equation}
\begin{equation}
0 \le p_m^{\mathrm c} \le z_m,
\qquad
0 \le p_m^{\mathrm h} \le g_m^{\mathrm h},
\qquad \forall m \in N,
\tag{M21}
\end{equation}
\begin{equation}
-u_{mn} \le p_{mn} \le u_{mn},
\qquad \forall (m,n) \in L,
\tag{M22}
\end{equation}
\begin{equation}
\tau p_m^{\mathrm c} = \rho \sum_{i \in I} f_{mi},
\qquad \forall m \in N,
\tag{M23}
\end{equation}
\begin{equation}
D_i = \sum_{m \in N} f_{mi},
\qquad \forall i \in I,
\tag{M24}
\end{equation}
\begin{equation}
\sum_{r \in R_t} w_{tr} + w_{t0} = 1,
\qquad
w_{tr} \le x_k,
\qquad \forall t \in T,\ \forall r \in R_t,\ \forall k \in S_{tr},
\tag{M25}
\end{equation}
\begin{equation}
l_r \ge \lambda_t - \sum_{k \in S_{tr}} \pi_{rk},
\qquad
\bar{l}_t \ge \lambda_t,
\qquad \forall t \in T,\ \forall r \in R_t,
\tag{M26}
\end{equation}
\begin{equation}
\sum_{r \in R_t} l_r w_{tr} + \bar{l}_t w_{t0}
\le
\lambda_t - \sum_{r \in R_t} \sum_{k \in S_{tr}} \psi_{rk},
\qquad \forall t \in T,
\tag{M27}
\end{equation}
\begin{equation}
\psi_{rk} \ge 0,\qquad
\psi_{rk} \le \pi_{rk},\qquad
\psi_{rk} \le M x_k,\qquad
\psi_{rk} \ge \pi_{rk} - M(1-x_k),
\quad \forall t \in T,\ \forall r \in R_t,\ \forall k \in S_{tr},
\tag{M28}
\end{equation}
\begin{equation}
x_i,\phi_{ij},u_i,v_i \in \{0,1\},
\qquad \forall i,j \in I,
\tag{M29}
\end{equation}
\begin{equation}
 \delta_{ij_1 j_2},\epsilon_{ij_1 j_2} \in \{0,1\},
\qquad \forall i,j_1,j_2 \in I,
\tag{M30}
\end{equation}
\begin{equation}
w_{tr},w_{t0} \in \{0,1\},
\qquad \forall t \in T,\ \forall r \in R_t,
\tag{M31}
\end{equation}
\begin{equation}
0 \le q_i \le 1,
\qquad
y_i,D_i \ge 0,
\qquad \forall i \in I,
\tag{M32}
\end{equation}
\begin{equation}
z_m,\ p_m^{\mathrm c},\ p_m^{\mathrm h},\ p_m^{\mathrm i} \ge 0,
\qquad \forall m \in N,
\tag{M33}
\end{equation}
\begin{equation}
u_{mn} \ge 0,\qquad f_{mi} \ge 0,\qquad \theta_m,\ p_{mn},\ \lambda_t \text{ free},
\qquad \forall (m,n) \in L,\ \forall m \in N,\ \forall i \in I,\ \forall t \in T,
\tag{M34}
\end{equation}
\begin{equation}
\pi_{rk} \ge 0,
\qquad \forall t \in T,\ \forall r \in R_t,\ \forall k \in S_{tr}.
\tag{M35}
\end{equation}
