\section*{符号表示法及输入数据}
让
\[
[N] := \{1,\ldots,N\},
\qquad N\in\mathbb{Z}_{++}.
\]
让$d_x,d_y\in\mathbb{Z}_{++}$保持不变。规定的样本和查询数据如下：
\[
x_0,\widehat{x}_i\in\mathbb{R}^{d_x},
\qquad
\widehat{y}_i\in\mathbb{R}^{d_y},
\qquad i\in[N].
\]
标量$\rho$、$\varepsilon$和$\eta$是规定的模型参数。这些规定的可行集合满足……
\[
\mathcal{A}\subseteq\mathbb{R}^{d_y},
\qquad
\mathcal{B}\subseteq\mathbb{R}.
\]
所有出现的$\|\cdot\|_2$都表示欧几里得范数。

\section*{决策变量}
决策变量是：
\[
\alpha\in\mathcal{A},
\qquad
\beta\in\mathcal{B},
\qquad
\lambda_1\in\mathbb{R}_+,
\qquad
\lambda_2\in\mathbb{R},
\]
\[
\theta\in\mathbb{R}_+^N,
\qquad
z\in\mathbb{R}_+^N,
\qquad
w\in\mathbb{R}.
\]

\section*{优化模型}
\begin{align}
\inf\quad
& \rho\lambda_1+\varepsilon\lambda_2
  +\frac{1}{N}\sum_{i\in[N]}\theta_i
\label{eq:objective}\\[0.25em]
\text{subject to}\quad
& \alpha\in\mathcal{A},
  \qquad \beta\in\mathcal{B},
\label{eq:sets}\\
& \lambda_1\ge 0,
  \qquad \lambda_2\in\mathbb{R},
\label{eq:lambda-domains}\\
& \theta_i\ge 0,
  \qquad z_i\ge 0,
  && \forall i\in[N],
\label{eq:vector-domains}\\
& 0\le w\le 1,
\label{eq:w-domain}\\
& \left\|
  \begin{bmatrix}
  2\alpha\\
  1-w-\varepsilon\lambda_1
  \end{bmatrix}
  \right\|_2
  \le 1-w+\varepsilon\lambda_1,
\label{eq:first-soc}\\
& z_i
  =\varepsilon\theta_i
  +\varepsilon\|x_0-\widehat{x}_i\|_2^2\lambda_1
  +\varepsilon\lambda_2
  +\frac{1}{4}\eta^2
  +\eta\beta,
  && \forall i\in[N],
\label{eq:z-definition}\\
& \left\|
  \begin{bmatrix}
  2\widehat{y}_i^{\top}\alpha-2\beta-\eta\\
  z_i-w
  \end{bmatrix}
  \right\|_2
  \le z_i+w,
  && \forall i\in[N].
\label{eq:second-soc}
\end{align}
