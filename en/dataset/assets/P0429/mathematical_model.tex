\section*{Sets and indices}
Let \(\mathcal N\) be a finite node set and let \(\mathcal E\subseteq\mathcal N\times\mathcal N\) be a finite directed-edge set. Let \(\mathcal E_h\subseteq\mathcal E\) be the set of candidate edges and \(\mathcal N_h\subseteq\mathcal N\) the set of candidate nodes. For every \(i\in\mathcal N_h\), let \(\mathcal E_h^{-}(i)\subseteq\mathcal E_h\) and \(\mathcal E_h^{+}(i)\subseteq\mathcal E_h\) denote the designated incoming and outgoing candidate-edge sets, respectively, and define the incident candidate-edge set
\[
\mathcal E_h(i):=\mathcal E_h^{-}(i)\cup\mathcal E_h^{+}(i).
\]
Let \(\mathcal S\subseteq\mathcal N\times\mathcal N\) be the finite set of origin--destination pairs, indexed by \((o,d)\). Let \(\mathcal R\) be the finite row-index set for the flow-balance equations.

\section*{Parameters}
\begin{itemize}
    \item \(q^{od}\in\mathbb R_+\): weight of OD pair \((o,d)\in\mathcal S\).
    \item \(\mathbf c=(c_{ij})_{(i,j)\in\mathcal E_h}\in\mathbb R^{|\mathcal E_h|}\): candidate-edge cost vector.
    \item \(\mathbf b=(b_i)_{i\in\mathcal N_h}\in\mathbb R^{|\mathcal N_h|}\): candidate-node cost vector.
    \item \(B_{\mathrm{edge}},B_{\mathrm{node}}\in\mathbb R\): edge and node budgets.
    \item \(\mathbf A\in\mathbb R^{|\mathcal R|\times|\mathcal E|}\): flow-balance matrix, whose columns are indexed by \(\mathcal E\).
    \item \(\mathbf e^{od}\in\mathbb R^{|\mathcal R|}\): flow-balance right-hand side for each \((o,d)\in\mathcal S\).
    \item \(\mathbf t=(t_{ij})_{(i,j)\in\mathcal E}\in\mathbb R^{|\mathcal E|}\): edge travel-time vector.
    \item \(\beta_1,\beta_2\in\mathbb R\): affine-piece slope coefficients satisfying \(\beta_1\leq\beta_2\).
    \item \(T_1\in\mathbb R\): breakpoint used to define the affine-piece intercepts
    \[
    \alpha_1:=1,
    \qquad
    \alpha_2:=1+(\beta_2-\beta_1)T_1.
    \]
\end{itemize}

\section*{Decision variables}
For every \((o,d)\in\mathcal S\), let
\[
\mathbf y^{od}=(y_{ij}^{od})_{(i,j)\in\mathcal E}\in\mathbb R^{|\mathcal E|}
\]
be the routing-flow vector and let \(v^{od}\in\mathbb R\) be its piecewise-linear performance value. For every \((i,j)\in\mathcal E_h\), let \(x_{ij}\) be the candidate-edge design variable, and for every \(i\in\mathcal N_h\), let \(z_i\) be the candidate-node design variable.

\section*{Mixed-integer linear formulation}
\begin{align}
\max_{\mathbf v,\mathbf x,\mathbf y,\mathbf z}\quad
& \sum_{(o,d)\in\mathcal S}q^{od}v^{od} \label{obj}\\
\text{subject to}\quad
& \sum_{(i,j)\in\mathcal E_h}c_{ij}x_{ij}
  \leq B_{\mathrm{edge}}, \label{edge-budget}\\
& \sum_{i\in\mathcal N_h}b_i z_i
  \leq B_{\mathrm{node}}, \label{node-budget}\\
& \mathbf A\mathbf y^{od}=\mathbf e^{od},
  && \forall (o,d)\in\mathcal S, \label{flow-balance}\\
& y_{ij}^{od}\leq x_{ij},
  && \forall (i,j)\in\mathcal E_h,\ \forall (o,d)\in\mathcal S, \label{edge-linking}\\
& y_{ij}^{od}\leq x_{wl}+z_i,
  && \forall i\in\mathcal N_h,\ \forall (i,j)\in\mathcal E_h^{-}(i),
     \ \forall (w,l)\in\mathcal E_h(i),\ \forall (o,d)\in\mathcal S, \label{node-linking}\\
& v^{od}\leq \alpha_1-\beta_1\sum_{(i,j)\in\mathcal E}t_{ij}y_{ij}^{od},
  && \forall (o,d)\in\mathcal S, \label{piece-one}\\
& v^{od}\leq \alpha_2-\beta_2\sum_{(i,j)\in\mathcal E}t_{ij}y_{ij}^{od},
  && \forall (o,d)\in\mathcal S, \label{piece-two}\\
& v^{od}\geq 0,
  && \forall (o,d)\in\mathcal S, \label{value-domain}\\
& 0\leq y_{ij}^{od}\leq 1,
  && \forall (i,j)\in\mathcal E,\ \forall (o,d)\in\mathcal S, \label{flow-domain}\\
& x_{ij}\in\{0,1\},
  && \forall (i,j)\in\mathcal E_h, \label{edge-domain}\\
& z_i\in\{0,1\},
  && \forall i\in\mathcal N_h. \label{node-domain}
\end{align}
