\section*{Shuttle assignment, trips and station times}
The requested pickup$e_p$,actual origin arrival$b_p$,and destination arrival$c_p$

The objective is the total cost expression (1a).
Let$B=B_L\dot\cup B_D\dot\cup B_T$denote idle,dwelling,travelling vehicles,
$P=P_N\dot\cup P_U$new and onboard groups,trip$k=1,\ldots,K$,stop$i=1,\ldots,I$.
All groups have one origin and one strictly later destination; their indicator
data are$\beta_{pi},\gamma_{pi}$and group size$z_p>0$.
Input includes current time$t_0$,current/next stop$\tilde i_b$,historical stop
arrival$h_{bi}$,remaining distance$r_b$,segment distance$d_i$,speed bounds,
capacity$C_b$,cost$f_b$,boarding/alighting seconds$a,e$,turnaround$\tau_I$,
maximum dwell$\tau_{\max}$,departure window$T_W$,and finite$M$.
Onboard group$p$has fixed$(\hat b_p,\hat k_p)$and historic pickup$h_p$.
Variables$\alpha_{bk},x_{bkp}\in\{0,1\}$,station arrival$A_{bki}$,
departure$D_{bki}$,load$Y_{bki}$,dwell$\delta_{bki}$,and$b_p,c_p,\Psi$
are nonnegative continuous except the binaries.
Define
\[
Q^+_{bki}=\sum_{p\in P}z_p\beta_{pi}x_{bkp},\quad
Q^N_{bki}=\sum_{p\in P_N}z_p\beta_{pi}x_{bkp},\quad
Q^-_{bki}=\sum_{p\in P}z_p\gamma_{pi}x_{bkp}.
\]
For$(b,k)$let$s_{bk}=\tilde i_b$if$k=1,b\in B_D\cup B_T$,otherwise1;
let$J_{bk}=\{s_{bk},\ldots,I\}$and
$Dwell_{bk}=\{s_{bk}+1,\ldots,I\}$if$k=1,b\in B_D$,otherwise$J_{bk}$.
Minimize
\[
w_1\sum_{p\in P_N}z_p(b_p-e_p)+w_2\sum_{p\in P}z_p(c_p-b_p)
+w_3\sum_{b,k}f_b\alpha_{bk}.
\]
The complete constraints implemented by the current code are
\begin{align*}
b_p-e_p&\le\Psi,\quad b_p\ge e_p&&p\in P_N,\\
b_p&=h_p,\quad c_p\ge t_0&&p\in P_U,\\
A_{b,1,1}&\ge\max(t_0,h_{bI}+\tau_I)&&b\in B_L,\\
r_b/v_b^{\max}&\le A_{b,1,\tilde i_b}-t_0\le r_b/v_b^{\min}&&b\in B_T,\\
A_{b,1,\tilde i_b}&=h_{b,\tilde i_b}&&b\in B_D,\\
D_{b,1,\tilde i_b}&\ge t_0,\quad
D_{b,1,\tilde i_b}\ge h_{b,\tilde i_b}+aQ^+_{b,1,\tilde i_b}&&b\in B_D,\\
D_{b,1,\tilde i_b}&\ge h_{b,\tilde i_b}+eQ^-_{b,1,\tilde i_b},\quad
D_{b,1,\tilde i_b}\le h_{b,\tilde i_b}+\tau_{\max}&&b\in B_D,\\
\alpha_{b,k+1}&\le\alpha_{bk}&&b\in B,k<K,\\
\alpha_{bk}&\le\sum_p x_{bkp}&&b\in B_L\text{ or }k>1,\\

\sum_{b,k}x_{bkp}&=1,\quad x_{bkp}\le\alpha_{bk}&&p\in P,\\

x_{b1p}&\le1-\sum_{i<\tilde i_b}\beta_{pi}&&b\in B_D\cup B_T,p\in P_N,\\
