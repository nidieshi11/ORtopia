\section*{MILP-Flow Reformulation}
The original SAHLP-h model contains cubic terms. The retained core model is the
flow-based MILP reformulation.

\begin{equation}
\begin{aligned}
\min \quad
& \sum_k f_k x_{kk}
+ \sum_i \sum_k c_{ik} x_{ik}
+ \sum_s \sum_k \sum_\ell d_{k\ell} \beta_{k\ell}^s z_{k\ell}^s
+ \sum_s \sum_k \sum_\ell \sum_i d_{k\ell} \alpha_{k\ell}^s y_{ik\ell}^s
\end{aligned}
\tag{M1}
\end{equation}

subject to
\begin{equation}
\sum_k x_{ik} = 1,
\qquad \forall i,
\tag{M2}
\end{equation}
\begin{equation}
x_{ik} \le x_{kk},
\qquad \forall i,k,
\tag{M3}
\end{equation}
\begin{equation}
\sum_i O_i x_{ik} \le \Gamma_k x_{kk},
\qquad \forall k,
\tag{M4}
\end{equation}
\begin{equation}
\sum_s z_{k\ell}^s \ge x_{kk} + x_{\ell\ell} - 1,
\qquad \forall k,\ell,\ k \neq \ell,
\tag{M5}
\end{equation}
\begin{equation}
\sum_s \sum_\ell y_{ik\ell}^s = O_i x_{ik},
\qquad \forall i,k,
\tag{M6}
\end{equation}
\begin{equation}
\sum_s \sum_\ell y_{i\ell k}^s = \sum_j w_{ij} x_{jk},
\qquad \forall i,k,
\tag{M7}
\end{equation}
\begin{equation}
\sum_i y_{ik\ell}^s \le U_{k\ell}^s z_{k\ell}^s,
\qquad \forall s,k,\ell,\ k \neq \ell,
\tag{M8}
\end{equation}
\begin{equation}
x_{ik} \in \{0,1\},
\qquad \forall i,k,
\tag{M9}
\end{equation}
\begin{equation}
z_{k\ell}^s \in \{0,1\},
\qquad \forall k,\ell,s,
\tag{M10}
\end{equation}
\begin{equation}
y_{ik\ell}^s \ge 0,
\qquad \forall i,k,\ell,s.
\tag{M11}
\end{equation}
