\section{Page 7}

\subsection{Objective Function}
\begin{equation}
\min \sum_{f\in F}\alpha_f y_f + \sum_{c\in C}\sum_{l\in L}\beta_{c,l}x_{c,l} \tag{1}
\end{equation}

\subsection{Constraints}
\begin{align}
& \text{s.t.} \quad \sum_{c\in C}\sum_{l\in L}\delta_{f,l}x_{c,l} + y_f = 1, \forall f \in F \tag{2} \\
& \sum_{c\in C}\sum_{l\in L}\delta_{m,l}x_{c,l} \leq 1, \forall m \in M \tag{3} \\
& \sum_{l\in L}\delta_{r,l}x_{c,l} = 1, \forall c \in C, r \in R_c \tag{4} \\
& \sum_{l\in L}x_{c,l} \leq 1, \forall c \in C \tag{5} \\
& \sum_{c\in C}\sum_{l\in L}\phi_{s,l}x_{c,l} \leq u_s, \forall s \in S \tag{6} \\
& x_{c,l} \in \{0, 1\}, \forall c \in C, l \in L \tag{7} \\
& y_f \in \{0, 1\}, \forall f \in F \tag{8}
\end{align}
