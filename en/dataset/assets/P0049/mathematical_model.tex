\section*{P0049 Placing Quantum Bits While Accounting for Noise}
\subsection*{Mathematical formulation}
\begin{equation}
\min \;
(1-\lambda)c_{\mathrm{swap}} + \lambda c_{\mathrm{noise}}
\tag{1a}
\end{equation}

\begin{equation}
c_{\mathrm{swap}} =
\sum_{t=1}^{N-1} \sum_{q\in Q} \sum_{i,j\in V\times V}
d_H(i,j)x_{q,i,j}^t
\tag{1b}
\end{equation}

\begin{equation}
c_{\mathrm{noise}} =
\sum_{i\in V} E_i \sum_{t=1}^N \sum_{q\in Q} w_{q,i}^t
+
\sum_{(i,j)\in A_H} E_{i,j} \sum_{t=1}^N \sum_{(p,q)\in L^t} z_{(p,q),(i,j)}^t
+
\sum_{(i,j)\in A_H} \sum_{k\in N((i,j))} E_{(i,j),k}
\sum_{t=1}^N \sum_{(r,s)\in L^t} \sum_{q\in Q}
u_{q,(r,s),(i,j),k}^t
\tag{1c}
\end{equation}

\begin{equation}
w_{q,i}^t = \sum_{j\in V} x_{q,i,j}^t,
\qquad
\forall 1 \le t \le N-1,\ \forall i \in V,\ \forall q \in Q
\tag{1d}
\end{equation}

\begin{equation}
w_{q,i}^{t+1} = \sum_{j\in V} x_{q,j,i}^t,
\qquad
\forall 1 \le t \le N-1,\ \forall i \in V,\ \forall q \in Q
\tag{1e}
\end{equation}

\begin{equation}
\sum_{i\in V} w_{q,i}^t = 1,
\qquad
\forall 1 \le t \le N,\ \forall q \in Q
\tag{1f}
\end{equation}

\begin{equation}
\sum_{q\in Q} w_{q,i}^t \le 1,
\qquad
\forall 1 \le t \le N,\ \forall i \in V
\tag{1g}
\end{equation}

\begin{equation}
\sum_{(i,j)\in A_H} z_{(p,q),(i,j)}^t = 1,
\qquad
\forall 1 \le t \le N,\ \forall (p,q) \in L^t
\tag{1h}
\end{equation}

\begin{equation}
\sum_{q\in Q} w_{q,i}^t = \sum_{q\in Q} w_{q,i}^1,
\qquad
\forall 2 \le t \le N,\ \forall i \in V
\tag{1j}
\end{equation}

\begin{equation}
\sum_{j:(i,j)\in A_H} z_{(p,q),(i,j)}^t = w_{p,i}^t,
\qquad
\forall 1 \le t \le N,\ \forall (p,q) \in L^t,\ \forall i \in V
\tag{2a}
\end{equation}

\begin{equation}
\sum_{i:(i,j)\in A_H} z_{(p,q),(i,j)}^t = w_{q,j}^t,
\qquad
\forall 1 \le t \le N,\ \forall (p,q) \in L^t,\ \forall j \in V
\tag{2b}
\end{equation}

\begin{equation}
u_{q,(r,s),(i,j),k}^t \le w_{q,k}^t,
\qquad
\forall t,(r,s),(i,j),k,q
\tag{2c}
\end{equation}

\begin{equation}
u_{q,(r,s),(i,j),k}^t \le z_{(r,s),(i,j)}^t,
\qquad
\forall t,(r,s),(i,j),k,q
\tag{2d}
\end{equation}

\begin{equation}
u_{q,(r,s),(i,j),k}^t \ge w_{q,k}^t + z_{(r,s),(i,j)}^t - 1,
\qquad
\forall t,(r,s),(i,j),k,q
\tag{2e}
\end{equation}

\begin{equation}
w_{q,i}^t \in \{0,1\},
\qquad
\forall 1 \le t \le N,\ \forall q \in Q,\ \forall i \in V
\tag{3a}
\end{equation}

\begin{equation}
x_{q,i,j}^t \in \{0,1\},
\qquad
\forall 1 \le t < N,\ \forall q \in Q,\ \forall (i,j) \in V \times V
\tag{3b}
\end{equation}

\begin{equation}
z_{(p,q),(i,j)}^t \in \{0,1\},
\qquad
\forall 1 \le t \le N,\ \forall (p,q) \in L^t,\ \forall (i,j) \in A_H
\tag{3c}
\end{equation}

\begin{equation}
u_{q,(r,s),(i,j),k}^t \in \{0,1\},
\qquad
\forall t,(r,s),(i,j),k,q
\tag{3d}
\end{equation}

\subsection*{Definitions and model scope}
A quantum-computing operator assigns the given logical quantum bits $Q$ to a connected directed hardware graph $H$ with hardware locations $V$ and hardware connections $A_H$. There are at least as many hardware locations as logical bits, and every hardware connection has its reverse. The circuit has a positive integer layer count $N$; the given two-bit gates $L^t$ in any layer share no logical bit. In every layer, each logical bit occupies exactly one hardware location, recorded by a yes-or-no placement choice $w_{q,i}^t$, and each hardware location holds at most one logical bit. The first layer's placement is chosen freely. At every hardware location, its total occupancy in each later layer must equal its occupancy in the first, so all layers use the same set of hardware locations.

Between each consecutive pair of layers, a yes-or-no transfer choice $x_{q,i,j}^t$ records whether a particular logical bit moves from one hardware location to another. Such choices exist for every ordered pair of hardware locations, including staying in the same location. For every logical bit and departure location, its transfer choices to all destinations sum exactly to its placement there in the earlier layer. For every destination location, its transfer choices from all origins sum exactly to its placement there in the next layer. The movement total $c_{\mathrm{swap}}$ adds the hardware shortest-path distance $d_H(i,j)$ times the transfer choice over all logical bits, all ordered location pairs and every layer transition. This measures movement distance; it does not itself prescribe a complete sequence of physical exchange gates.

Every two-bit gate in every layer must be assigned to exactly one directed hardware connection through a yes-or-no gate assignment $z_{(p,q),(i,j)}^t$. For that gate, assignments on connections leaving a particular location must sum to the first logical bit's placement at that location; assignments on connections entering a particular location must sum to the second logical bit's placement there. Thus both gate endpoints match their ordered logical-bit placements and the gate uses a hardware connection. A connection's noise cost is given for its actual direction; reverse-direction costs are not presumed equal.

The given single-bit and readout costs $E_i$ and directed two-bit gate costs $E_{i,j}$ are nonnegative. For a hardware connection, take each endpoint's hardware neighbours $N_H(i)$, remove the other endpoint from each collection, and unite them as its nearby spectator locations $N((i,j))$. At every such spectator location, the supplied measured error-rate ratio $r_{(i,j)}^k$ is nonnegative. The crosstalk cost $E_{(i,j),k}$ is the larger of zero and the measured ratio minus one multiplied by that spectator location's single-bit cost; the entire ratio-minus-one difference is multiplied by the cost. For each layer, each gate of that layer, each hardware connection, each of its spectator locations and every logical bit, a yes-or-no joint exposure record $u_{q,(r,s),(i,j),k}^t$ cannot exceed the logical bit's placement at the spectator or the gate assignment on the connection, and must be at least their sum minus one. It is therefore one exactly when both events occur.

The total noise charge $c_{\mathrm{noise}}$ first sums each hardware location's single-bit cost for every logical bit placed there in every layer, irrespective of whether that bit participates in a two-bit gate. It then adds each directed connection's two-bit cost for every gate assigned there in every layer. Finally it adds each connection-spectator crosstalk cost for every joint exposure record, over all layers, gates, spectators and logical bits in their stated ranges. The operator seeks the lowest combined charge: the movement total multiplied by the difference between one and the given trade-off weight $\lambda$, plus that weight times the noise charge. The weight lies between zero and one, including both endpoints. Placement, transfer, gate assignment and exposure records all take only zero or one. Hardware distances and all applicable noise measurements must be supplied, and missing costs are not treated as zero.
