\section*{Stochastic neoadjuvant chemotherapy MILP (17)}

Let $\mathcal D$ be the set of available drugs, $\mathcal Q$ the set of cancer cell types, and $\mathcal S=\{0,\ldots,S\}$ the time indices. For fixed step size $h$, the grid satisfies $t(0)=0$, $t(S)=T=Sh$. The scenario set is indexed by $k\in\{1,\ldots,K\}$. The model is
\begin{align}
\min\quad &\sum_{q\in\mathcal Q}P_{q,S}^{(1)} \tag{17a}\\
\text{s.t.}\quad
&C_{d,s+1}=C_{d,s}-h\xi_d C_{d,s}+U_{d,s}/\mathcal V,
&&\forall d\in\mathcal D,\ s\in\{0,\ldots,S-1\}, \tag{17b}\\
&C_{d,0}=0, &&\forall d\in\mathcal D, \tag{17c}\\
&P_{q,s+1}^{(k)}=P_{q,s}^{(k)}+h\left(\Lambda\left(\ln(N_{q,\infty}^{(k)})-P_{q,s}^{(k)}\right)
-\sum_{d\in\mathcal D}\eta_{d,q}\exp(-\rho_{d,q}t(s))E_{d,s}\right),
&&\forall q\in\mathcal Q,\ s\in\{0,\ldots,S-1\},\ k\in\{1,\ldots,K\}, \tag{17d}\\
&P_{q,0}^{(k)}=\ln(N_{q,0}^{(k)}),
&&\forall q\in\mathcal Q,\ k\in\{1,\ldots,K\}, \tag{17e}\\
&E_{d,s}\geq 0, &&\forall d\in\mathcal D,\ s\in\mathcal S, \tag{10a}\\
&E_{d,s}\geq C_{d,s}-\beta_{d,\mathrm{eff}}, &&\forall d\in\mathcal D,\ s\in\mathcal S, \tag{10b}\\
&E_{d,s}\leq\beta_{d,\mathrm{conc}}Z_{E,d,s}, &&\forall d\in\mathcal D,\ s\in\mathcal S, \tag{10c}\\
&E_{d,s}\leq C_{d,s}-\beta_{d,\mathrm{eff}}+\beta_{d,\mathrm{conc}}(1-Z_{E,d,s}),
&&\forall d\in\mathcal D,\ s\in\mathcal S, \tag{10d}\\
&Z_{E,d,s}\in\mathbb B, &&\forall d\in\mathcal D,\ s\in\mathcal S, \tag{10e}\\
&P_{q,S}^{(k)}\leq P_{\mathrm{surg}}+
\ln\left(\frac{N_{q,0}^{(k)}}{\sum_{q\in\mathcal Q}N_{q,0}^{(k)}}\right)
+P_{q,\infty}(1-Z_{\mathrm{surg}}^{(k)}),
&&\forall q\in\mathcal Q,\ k\in\{1,\ldots,K\}, \tag{17g}\\
&\sum_{k=1}^{K}\mu^{(k)}Z_{\mathrm{surg}}^{(k)}\geq 1-\epsilon, && \tag{17h}\\
&Z_{\mathrm{surg}}^{(k)}\in\mathbb B, &&\forall k\in\{1,\ldots,K\}, \tag{17i}\\
&(\mathbf U,\mathbf C,\mathbf N_{\mathrm{neu}},\mathbf N_{\mathrm{lym}})\in\widehat{\mathcal O}. && \tag{17j}
\end{align}

\paragraph{Scenario and parameter definitions.}
The tumor heterogeneity is represented by a discrete random vector
\[
\pi\in\mathbb R_+^{|\mathcal Q|},\qquad
\pi\in\{\pi^{(1)},\ldots,\pi^{(K)}\},\qquad
\mu^{(k)}=\Pr\{\pi=\pi^{(k)}\},\quad k\in\{1,\ldots,K\},
\]
with the stated most-likely-scenario premise
\[
\mu^{(1)}\geq\mu^{(k)},\qquad \forall k\in\{1,\ldots,K\}.
\]
The source defines $P_{\mathrm{surg}}=\ln(N_{\mathrm{surg}})$ and $P_{q,\infty}=\ln(N_{q,\infty})$. Here $N_{\mathrm{surg}}$ is the fixed clinically determined operable size and $\epsilon$ is the fixed probability of not meeting the target. The superscript $(k)$ denotes the value under scenario $k$; the treatment variables $U,C,E$ are shared across scenarios. The objective minimizes terminal log populations for scenario 1, not an expectation over scenarios.

The fixed effect-compartment volume satisfies $\mathcal V>0$. The variables $U_{d,s}$, $C_{d,s}$, and $E_{d,s}$ denote administered dose, concentration, and effective concentration, respectively. The logarithmic population state has the meaning $P_{q,s}^{(k)}=\ln(N_{q,s}^{(k)})$. The parameters $\xi_d,\Lambda,\eta_{d,q},\rho_{d,q}$ are respectively the drug elimination rate, Gompertz shape parameter, cancer-cell fractional kill effect, and temporal effectiveness decay parameter. The paper states $\eta_{d,q}\geq0$. The fixed input populations $N_{q,0}^{(k)}$ and $N_{q,\infty}^{(k)}$ are scenario initial and asymptotic cell-type populations.

\paragraph{Selected operational-set construction.}
In (17j), choose the paper's discrete approximation (13), rather than the alternative McCormick constraints (12). Write $N_{w,0}$ for the fixed pretreatment white-blood-cell count, $N_{w,s}$ for its discretized count, and $B_{d,s}$ for the approximation of the product of white-blood-cell count and delayed drug concentration. The paper assumes
\[
\beta_w=\min\left\{\frac{\beta_{\mathrm{neu}}}{\theta_{\mathrm{neu}}},
\frac{\beta_{\mathrm{lym}}}{\theta_{\mathrm{lym}}}\right\},\qquad
0\leq C_{d,s}\leq\beta_{d,\mathrm{conc}},\quad
\beta_w\leq N_{w,s}\leq N_{w,0},
\]
for all applicable $d\in\mathcal D$ and $s\in\mathcal S$. The delay index $\tau$ corresponds to the response delay $t_w$. Before drug effects begin, the unnumbered recurrence printed on page 8 is
\[
N_{w,s+1}=N_{w,s}+h(\upsilon_w-\nu_wN_{w,s}),
\qquad \forall s\in\{0,\ldots,\tau-1\},
\]
where $N_{w,0}$ is the initial count. The selected post-delay construction is
\begin{align}
N_{w,s+1}&=N_{w,s}+h\left(\upsilon_w-\nu_wN_{w,s}
-\sum_{d\in\mathcal D}\eta_{d,w}B_{d,s}\right),
&&\forall s\in\{\tau,\ldots,S-1\}, \tag{13a}\\
-\frac{\Delta}{2}\leq N_{w,s}-\left(\beta_w+\sum_{k=1}^{K}k\Delta Z_{w,s,k}\right)\leq\frac{\Delta}{2},
&&\forall s\in\{\tau,\ldots,S-1\}, \tag{13b--c}\\
&\sum_{k=0}^{K}Z_{w,s,k}=1, &&\forall s\in\{\tau,\ldots,S-1\}, \tag{13d}\\
&B_{d,s}=\sum_{k=1}^{K}(\beta_w+k\Delta)V_{d,s,k},
&&\forall d\in\mathcal D,\ s\in\{\tau,\ldots,S-1\}, \tag{13e}\\
&V_{d,s,k}\leq\beta_{d,\mathrm{conc}}Z_{w,s,k},
&&\forall d\in\mathcal D,\ s\in\{\tau,\ldots,S-1\},\ k\in\{1,\ldots,K\}, \tag{13f}\\
&V_{d,s,k}\leq C_{d,s-\tau},
&&\forall d\in\mathcal D,\ s\in\{\tau,\ldots,S-1\},\ k\in\{1,\ldots,K\}, \tag{13g}\\
&V_{d,s,k}\geq C_{d,s-\tau}+\beta_{d,\mathrm{conc}}(Z_{w,s,k}-1),
&&\forall d\in\mathcal D,\ s\in\{\tau,\ldots,S-1\},\ k\in\{1,\ldots,K\}, \tag{13h}\\
&V_{d,s,k}\geq0,
&&\forall d\in\mathcal D,\ s\in\{\tau,\ldots,S-1\},\ k\in\{1,\ldots,K\}, \tag{13i}\\
&Z_{w,s,k}\in\mathbb B,
&&\forall s\in\{\tau,\ldots,S-1\},\ k\in\{1,\ldots,K\}. \tag{13j}
\end{align}
The two inequalities in (13b--c) are the printed pair
\[
N_{w,s}-\left(\beta_w+\sum_{k=1}^{K}k\Delta Z_{w,s,k}\right)\leq\frac{\Delta}{2},\qquad
-N_{w,s}+\left(\beta_w+\sum_{k=1}^{K}k\Delta Z_{w,s,k}\right)\leq\frac{\Delta}{2}.
\]
The prose introducing the discrete approximation on page 7 specifies $Z_{w,s,k}\in\mathbb B$ for $k\in\{0,\ldots,K\}$; include this domain for all $k=0,\ldots,K$. The printed (13j) gives the narrower range $k\in\{1,\ldots,K\}$, which conflicts with that prose; retain the printed (13j) scope and note the conflict.

The neutropenia and lymphocytopenia constraints included in the operational set are
\begin{align*}
N_{\mathrm{neu},s}&\geq\beta_{\mathrm{neu}},\qquad
N_{\mathrm{neu},s}=\theta_{\mathrm{neu}}N_{w,s},
&&\forall s\in\{0,\ldots,S\},\\
N_{\mathrm{lym},s}&\geq\beta_{\mathrm{lym}},\qquad
N_{\mathrm{lym},s}=\theta_{\mathrm{lym}}N_{w,s},
&&\forall s\in\{0,\ldots,S\}.
\end{align*}
Here $\upsilon_w$ is white-blood-cell production, $\nu_w$ is turnover, $\eta_{d,w}$ is the white-blood-cell fractional kill effect, and $\theta_{\mathrm{neu}},\theta_{\mathrm{lym}}$ are the fixed fractions of total white blood cells. The fixed clinical thresholds are $\beta_{\mathrm{neu}},\beta_{\mathrm{lym}}$. The drug effect on white blood cells uses concentration, including below $\beta_{d,\mathrm{eff}}$, and does not decay through temporal resistance.

\paragraph{Discrete operational administration constraints.}
The operational-constraint definition partitions the treatment period into $M$ days $D_m$, with $\mathrm{DAYS}=\{1,\ldots,M\}$, and designates meal times by $\mathrm{MEALS}$. The detailed official Appendix A rows specialize to capecitabine, docetaxel, and etoposide, labeled $1,2,3$. For capecitabine:
\begin{align*}
C_{1,s}&\leq\beta_{1,\mathrm{conc}}, &&\forall s\in\mathcal S,\\
U_{1,s}&\leq\beta_{1,\mathrm{rate}}, &&\forall s\in\mathcal S,\\
\sum_{s\in D_m}U_{1,s}&\leq\beta_{1,\mathrm{cum}}, &&\forall m\in\mathrm{DAYS},\\
U_{1,s}&=0, &&\forall s\notin\mathrm{MEALS},\\
U_{1,s}&=\alpha_{1,\mathrm{pill}}Z_{1,\mathrm{pill},s}, &&\forall s\in\mathrm{MEALS},\\
Z_{1,\mathrm{pill},s}&\in\mathbb Z_+, &&\forall s\in\mathrm{MEALS}.
\end{align*}
For docetaxel:
\begin{align*}
C_{2,s}&\leq\beta_{2,\mathrm{conc}}, &&\forall s\in\mathcal S,\\
U_{2,s}&\leq\beta_{2,\mathrm{rate}}, &&\forall s\in\mathcal S,\\
\sum_{s\in D_m}U_{2,s}&\leq\beta_{2,\mathrm{cum}}(1-Z_{2,\mathrm{rest}}^m), &&\forall m\in\mathrm{DAYS},\\
\sum_{l=0}^{\min\{\alpha_{2,\mathrm{rest}},M-m\}}(1-Z_{2,\mathrm{rest}}^{m+l})&\leq1, &&\forall m\in\mathrm{DAYS},\\
Z_{2,\mathrm{rest}}^m&\in\mathbb B, &&\forall m\in\mathrm{DAYS}.
\end{align*}
For etoposide:
\begin{align*}
C_{3,s}&\leq\beta_{3,\mathrm{conc}}, &&\forall s\in\mathcal S,\\
U_{3,s}&\leq\beta_{3,\mathrm{rate}}, &&\forall s\in\mathcal S,\\
\sum_{s\in D_m}U_{3,s}&\leq\beta_{3,\mathrm{cum}}, &&\forall m\in\mathrm{DAYS},\\
U_{3,s}&=0, &&\forall s\notin\mathrm{MEALS},\\
U_{3,s}&=\alpha_{3,\mathrm{pill}}Z_{3,\mathrm{pill},s}, &&\forall s\in\mathrm{MEALS},\\
Z_{3,\mathrm{pill},s}&\in\mathbb Z_+, &&\forall s\in\mathcal S.
\end{align*}
The source does not specify the exact meal-time indices in these mathematical rows. The docetaxel rest-sum endpoint is inclusive as printed. The rest rows apply to the stated rest-requiring drug; pill rows apply to the stated oral drugs.

The operational set also enforces nonnegativity of $U_{d,s},C_{d,s},E_{d,s},N_{w,s},N_{\mathrm{neu},s},N_{\mathrm{lym},s}$, except that cancer log-population states $P_{q,s}^{(k)}$ need not be nonnegative. The elected construction has $B_{d,s}$ as an approximation variable and $V_{d,s,k}\geq0$ as specified above. The parameter premises stated for the underlying models include $\mathcal V>0$; the paper's Euler stability results additionally state $\xi_d,h,\mathcal V>0$ and $h<2/\xi_d$, and $\Lambda,h>0$ and $h<2/\Lambda$. These stability conditions are premises of those analytical results, not extra rows of (17).

\paragraph{Printed coefficient distinction.}
Equation (17d) prints the scenario-indexed $N_{q,\infty}^{(k)}$, whereas (17g) prints the scenario-independent $P_{q,\infty}$, defined as $\ln(N_{q,\infty})$. Both printed forms are retained; the coefficient in (17g) is not changed to a scenario-indexed quantity. The chance-constraint surrogate (17g)--(17i) is the selected conservative construction; the original probability constraint is not imposed as an additional row.
