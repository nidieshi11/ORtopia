\section{P-Path Reformulation}

\begin{equation}
\min \quad
\sum_{h,l\in H} \beta_{hl} z_{hl}
+ \sum_{r\in T_{\mathrm{core}}} p^r g^r
+ \sum_{r\in T_{\mathrm{latent}}} \sum_{\pi\in A^r} p^r \lambda_\pi^r \bigl(g(\pi)-\varphi\bigr)
\tag{15a}
\end{equation}

\begin{equation}
\begin{aligned}
\text{s.t.}\quad
& \sum_{l\in H} z_{hl} = \sum_{l\in H} z_{lh}
&& \forall h\in H, \\
& z_{hl} = 1
&& \forall (h,l)\in Z_{\mathrm{fixed}}, \\
& g^r = \sum_{h,l\in H} \tau_{hl}^r x_{hl}^r + \sum_{i,j\in N} \gamma_{ij}^r y_{ij}^r
&& \forall r\in T, \\
& \sum_{h\in H:\, i\in H} (x_{ih}^r - x_{hi}^r)
+ \sum_{j\in N} (y_{ij}^r - y_{ji}^r)
=
\begin{cases}
1, & i = or^r, \\
-1, & i = de^r, \\
0, & \text{otherwise},
\end{cases}
&& \forall i\in N,\ \forall r\in T, \\
& x_{hl}^r \le z_{hl}
&& \forall h,l\in H,\ \forall r\in T, \\
& f_\pi^r \le z_{hl}
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r\cup RP^r,\ \forall (h,l)\in x(\pi), \\
& \sum_{(h,l)\in x(\pi)} z_{hl} - |x(\pi)| + 1 \le f_\pi^r
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r\cup RP^r, \\
& g^r \le g(\pi) + M_\pi^r (1-f_\pi^r)
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r\cup RP^r, \\
& \lambda_\pi^r \le f_\pi^r
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r, \\
& \lambda_\pi^r \le 1 - x_{hl}^r
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r,\ \forall (h,l)\notin x(\pi), \\
& \lambda_\pi^r \ge
\sum_{(h,l)\in x(\pi)} x_{hl}^r
- |x(\pi)|
- \sum_{(h,l)\notin x(\pi)} x_{hl}^r + 1
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r, \\
& z_{hl},\ x_{hl}^r,\ y_{ij}^r \in \{0,1\}
&& \forall h,l\in H,\ \forall i,j\in N,\ \forall r\in T, \\
& f_\pi^r \in \{0,1\}
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r\cup RP^r, \\
& \lambda_\pi^r \in \{0,1\}
&& \forall r\in T_{\mathrm{latent}},\ \forall \pi\in A^r, \\
& g^r \ge 0
&& \forall r\in T.
\end{aligned}
\end{equation}
