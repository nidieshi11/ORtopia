\section*{Supplier trade-credit contract optimization}
\begin{equation}
\begin{aligned}
\max_{w_c,w_t}\quad \Pi_s
&=-cq^w+w_cq_c^w
 +(1-\alpha_t)\int_{\theta_b^w}^{\theta_t^w}(x-\theta_b^w)\,dF(x)
 +(\theta_t^w-\theta_b^w)\bar F(\theta_t^w),\\
\text{subject to}\quad &c\leq w_c\leq w_t\leq 1,\\
&(q^w,q_c^w,\theta_b^w,\theta_t^w)
 \text{ are the retailer's best response specified below.}
\end{aligned}\tag{7}
\end{equation}
The maximization and subject-to presentation combines the supplier's objective in Section 5 with the contract domain in Section 3.1 and its stated dependence on Proposition 1. The selected model is the full contract-design problem, including the response without external financing; it is not restricted to the external-financing expression (8).

\section*{Fixed inputs, decisions, and timing}
There is one supplier, one retailer, and one product. The fixed unit production cost is $c$. Retail price, salvage price, risk-free rate, initial retailer cash, and default-cost proportions satisfy
\[
p=1,\qquad s=0,\qquad r_f=0,\qquad K>0,\qquad
\alpha_b,\alpha_t\in[0,1].
\]
The demand distribution is fixed, with stochastic demand $\widetilde D$ supported on $[0,+\infty)$, realization $D$, CDF $F$, and density $f$. Its associated functions are
\[
\bar F(x)=1-F(x),\qquad h(x)=\frac{f(x)}{\bar F(x)},\qquad
 g(x)=xh(x).
\]
The distribution has an increasing failure rate (IFR). Unmet demand is lost. The retailer can use only future revenue to repay external liabilities. Except for default costs, the financial market is perfectly competitive; parties are risk-neutral, have symmetric information, and face no moral hazard.

The supplier first offers the cash and credit prices $(w_c,w_t)$. The retailer then selects the total order and its financing allocation; delivery and cash payment precede demand realization, and repayment follows demand realization. The retailer's quantity domains and split are
\[
q\geq0,\qquad q_c\in[0,q],\qquad q_t=q-q_c.
\]
Here $q_c$ is inventory purchased at the cash price, not necessarily inventory financed solely with internal cash; $q_t$ is inventory purchased on trade credit. Cash payment is $w_cq_c$, and nominal trade credit is $w_tq_t$. The early-payment discount is
\[
d_t=1-\frac{w_c}{w_t}.
\]
Net terms mean $w_c=w_t$; two-part terms mean $w_c<w_t$. Superscript $w$ denotes the retailer's response to a fixed contract; superscript $*$ denotes an endogenous optimal contract and its induced quantities.

\paragraph{Cash-domain discrepancy.}
Section 3 specifies $K>0$, retained here as the input premise. Appendix A lists $K\geq0$, and Proposition 1 prints intervals starting at zero. Those intervals are transcribed below without weakening the Section 3 premise.

\section*{Financing and retailer payoff definitions}
The bank loan principal and repayment threshold are
\[
B=(w_cq_c-K)^+,\qquad \theta_b=(1+r_b)B,
\]
where $r_b$ is the bank interest rate. The bank loan is senior to trade credit. The bank loan is fully repaid if and only if $D\geq\theta_b$; trade credit defaults when $D<\theta_t$. The trade-credit threshold is
\begin{equation}
\theta_t=\theta_b+w_t(q-q_c).\tag{1}
\end{equation}
The proportions $\alpha_b$ and $\alpha_t$ are deadweight costs proportional to the values seized upon default. A zero proportion means that the corresponding creditor incurs no recovery deadweight loss.

The end-of-horizon payoffs printed in Table 2 are
\[
\begin{array}{c|cccc}
& D\in[0,\theta_b)&D\in[\theta_b,\theta_t)&D\in[\theta_t,q]&D>q\\ \hline
\text{Bank}&(1-\alpha_b)D&\theta_b&\theta_b&\theta_b\\
\text{Supplier}&0&(1-\alpha_t)(D-\theta_b)&\theta_t-\theta_b&\theta_t-\theta_b\\
\text{Retailer}&0&0&D-\theta_t&q-\theta_t
\end{array}
\]
These are terminal payoffs, not the supplier's total profit including production and initial cash sales.

\paragraph{Cash not exhausted: $w_cq_c<K$.}
The retailer uses neither a bank loan nor trade credit; $q_t=0$. The retailer maximizes
\[
\Pi_r=\int_0^q\bar F(x)\,dx-w_cq.
\]

\paragraph{Cash exhausted: $w_cq_c\geq K$.}
Competitive bank pricing requires
\[
\int_0^{\theta_b}(1-\alpha_b)x\,dF(x)
 +\theta_b\bar F(\theta_b)=B=w_cq_c-K.
\]
Define
\[
\hat F_b(x):=\bar F(x)[1-\alpha_bg(x)].
\]
Then the bank pricing relation and the printed interest-rate expression are
\begin{equation}
w_cq_c-K-\int_0^{\theta_b}\hat F_b(x)\,dx=0,\tag{2}
\end{equation}
\begin{equation}
r_b=\frac{\theta_b-B}{B}
=\frac{\theta_b}{\displaystyle\int_0^{\theta_b}\hat F_b(x)\,dx}-1.\tag{3}
\end{equation}
The retailer chooses $q_c$ and $q$, equivalently $q_c$ and $q_t$, to maximize
\begin{equation}
\Pi_r=\int_{\theta_t}^{q}\bar F(x)\,dx-K\tag{4}
\end{equation}
subject to (1), (2), and the quantity domains above, within this cash-exhausted case. Equation (3) is a loan-pricing expression; the paper supplies no value for its zero-principal division at $B=0$.

\section*{Complete printed retailer response: Proposition 1}
Define, exactly as printed at the start of Proposition 1,
\[
\kappa^{nb}(w_c)=w_c\bar F^{-1}(w_c),
\]
\[
\kappa_r^b(w_c,w_t)
=w_c\bar F^{-1}\!\left(
\frac{w_c}{1-\alpha_bg\bigl(\hat F_b^{-1}(w_c/w_t)\bigr)}
\right)
-\int_0^{\hat F_b^{-1}(w_c/w_t)}\hat F_b(x)\,dx,
\]
\[
C(\theta)=\frac{K+\displaystyle\int_0^{\theta}\hat F_b(x)\,dx}
{\hat F_b(\theta)}-\theta.
\]
The three response branches are as follows, with $q_t^w=q^w-q_c^w$ and the quantity domains inherited from Appendix A.

\paragraph{1. No external financing.}
For $K\in[\kappa^{nb}(w_c),+\infty)$,
\[
q^w=q_c^w=\bar F^{-1}(w_c),\qquad
\theta_b^w=\theta_t^w=0.
\]

\paragraph{2. Bank loan only.}
For $K\in[\kappa^b(w_c,w_t),\kappa^{nb}(w_c))$,
\[
q^w=q_c^w=\bar F^{-1}\!\left(
\frac{w_c}{1-\alpha_bg(\theta_b^w)}\right),
\qquad \theta_t^w=\theta_b^w,
\]
where the threshold is uniquely determined by
\begin{equation}
w_c\bar F^{-1}\!\left(\frac{w_c}{1-\alpha_bg(\theta_b^w)}\right)
-\int_0^{\theta_b^w}\hat F_b(x)\,dx=K.\tag{5}
\end{equation}

\paragraph{3. Bank loan and trade credit.}
For $K\in[0,\kappa^b(w_c,w_t))$,
\[
\theta_b^w=\hat F_b^{-1}(w_c/w_t),\qquad
q_c^w=\frac{K+\displaystyle\int_0^{\theta_b^w}\hat F_b(x)\,dx}{w_c},
\]
and $(q^w,\theta_t^w)$ jointly satisfy
\[
w_t\bar F(\theta_t^w)=\bar F(q^w)
\]
and
\begin{equation}
[\theta_t^w+C(\theta_b^w)]\bar F(\theta_t^w)
=q^w\bar F(q^w).\tag{6}
\end{equation}

\paragraph{Printed threshold-name discrepancy.}
The definition in Proposition 1 has the subscript in $\kappa_r^b$, whereas its branch intervals and the subsequent proofs use $\kappa^b$. The source does not separately define the latter. Both printed forms have been retained rather than silently introducing an equality between them. Likewise, the third branch's verbal description does not exclude a zero bank threshold: its printed formula yields $\theta_b^w=\hat F_b^{-1}(1)$ under net terms.

\paragraph{Supporting response condition.}
Appendix C, Lemma C.3, states that, for a given $(w_c,w_t)$, a response $(q_c^{w,3},q_t^{w,3},\theta_b^{w,3},\theta_t^{w,3})$ with $q_t^{w,3}>0$ is globally optimal for
$\Pi_r=\int_{\theta_t}^{q_c+q_t}\bar F(x)\,dx$ subject to (2) and (1) if and only if both
\begin{equation}
\hat F_b(\theta_b^{w,3})=\frac{w_c}{w_t},\tag{69}
\end{equation}
\begin{equation}
w_t\bar F(\theta_t^{w,3})=\bar F(q_c^{w,3}+q_t^{w,3})\tag{70}
\end{equation}
hold. This is the supporting characterization of the third branch, not another supplier optimization problem. Appendix C states that the technical-lemma proofs are available upon request; those proofs are unavailable here and remain unverified.

\section*{Conditional threshold representation}
When external financing is used, Section 5 uses
\[
w_cq_c^w=K+\int_0^{\theta_b^w}\hat F_b(x)\,dx
\]
to write the same supplier profit as
\begin{equation}
\begin{aligned}
\Pi_s={}&K+\int_0^{\theta_t^w}\bar F(x)\,dx-cQ(\theta_b^w,\theta_t^w)\\
&-\left[\alpha_b\int_0^{\theta_b^w}x\,dF(x)
+\alpha_t\int_{\theta_b^w}^{\theta_t^w}(x-\theta_b^w)\,dF(x)\right].
\end{aligned}\tag{8}
\end{equation}
The referenced set and quantity definition from Corollary 3 are
\[
\Theta:=\left\{(\theta_b,\theta_t)\;\middle|\;
\begin{aligned}
&\theta_t\geq\theta_b\geq0,\\
&[\theta_t+C(\theta_b)]\bar F(\theta_t)
 \leq g^{-1}(1)\bar F\bigl(g^{-1}(1)\bigr)
\end{aligned}\right\},
\]
\[
(\theta_b^w,\theta_t^w)\in\Theta,
\qquad
q^w=Q(\theta_b^w,\theta_t^w)
:=\min\left\{q:q\bar F(q)
 =[\theta_t^w+C(\theta_b^w)]\bar F(\theta_t^w)\right\}.
\]
Corollary 3 claims a one-to-one mapping to prices satisfying $w_c\leq w_t\leq1$. The original contract restriction $c\leq w_c$ still belongs to the full-design model; it is not printed in the definition of $\Theta$. The proof selects the smaller root because (8) decreases in $q$, and states that this selection guarantees $g(q)\leq1$.

\paragraph{Scope and mapping discrepancies.}
Equation (8) is printed only under external financing and does not replace (7) for the no-external-financing response. Corollary 3's unrestricted one-to-one claim cannot include the entire first branch of Proposition 1, in which different contracts can give $(\theta_b^w,\theta_t^w)=(0,0)$. Its proof also prints $1-w_c/w_t=\hat F_b(\theta_b^w)$, conflicting with (69), the third branch of Proposition 1, and (11) below. The operative response chain here retains (69), not that conflicting proof expression. No additional restriction on $\Theta$ or root convention has been supplied to repair these claims.

\section*{Linked optimality characterization, not additional feasibility constraints}
Equations (9)--(13) are derivatives and necessary optimality conditions for the threshold analysis; they are not a final replacement subject-to block for the full-design problem. They are recorded with their printed scope and are not imposed on every feasible contract.

\begin{equation}
\frac{\partial\Pi_s}{\partial\theta_t^w}
=\bar F(\theta_t^w)-c\frac{\partial Q}{\partial\theta_t^w}
-\alpha_t(\theta_t^w-\theta_b^w)f(\theta_t^w).\tag{9}
\end{equation}
\begin{equation}
\frac{\partial\Pi_s}{\partial\theta_b^w}
=-c\frac{\partial Q}{\partial\theta_b^w}
+\left\{[\alpha_t-\alpha_bg(\theta_b^w)]\bar F(\theta_b^w)
-\alpha_t\bar F(\theta_t^w)\right\}.\tag{10}
\end{equation}
Lemma 1 gives the optimal-price expressions
\begin{equation}
w_c^*=\frac{\hat F_b(\theta_b^*)\bar F(q^*)}{\bar F(\theta_t^*)},
\qquad
w_t^*=\frac{\bar F(q^*)}{\bar F(\theta_t^*)},\tag{11}
\end{equation}
where $q^*=Q(\theta_b^*,\theta_t^*)$ and the printed alternatives are:
\begin{equation}
\theta_b^*=0,\qquad
\frac{c}{\bar F(q^*)[1-g(q^*)]}
=\frac{1-\alpha_tg(\theta_t^*)}{1-(\theta_t^*+K)h(\theta_t^*)}
\geq\frac{\alpha_t[1-\bar F(\theta_t^*)]}{f(0)K\bar F(\theta_t^*)},\tag{12}
\end{equation}
or
\begin{equation}
\begin{gathered}
\theta_b^*>0,\\
\frac{c}{\bar F(q^*)[1-g(q^*)]}
=\frac{1-\alpha_t(\theta_t^*-\theta_b^*)h(\theta_t^*)}
{1-[\theta_t^*+C(\theta_b^*)]h(\theta_t^*)}
=\frac{[\alpha_t-\alpha_bg(\theta_b^*)]\bar F(\theta_b^*)
-\alpha_t\bar F(\theta_t^*)}{C'(\theta_b^*)\bar F(\theta_t^*)}.
\end{gathered}\tag{13}
\end{equation}
Here $C'$ is the derivative of the previously defined $C$. The lemma states necessity, not general sufficiency. Its proof on physical PDF page 37 prints the strict condition $\partial\Pi_s/\partial\theta_b\vert_{(0,\theta_t^*)}<0$ while calling it equivalent to the weak inequality in (12); (12) is retained as printed. No assumption $f(0)>0$ or zero-denominator convention is added to (12).
